001  (ns com.ruoyi.web.handler
002    (:require
003     [clojure.string :as str]
004     [com.ruoyi.web.middleware.core :as middleware]
005     [integrant.core :as ig]
006     [ring.util.response :as response]
007     [reitit.ring :as ring]
008     [reitit.swagger-ui :as swagger-ui]))
009  
010  (defn- spa-not-found-handler
011    "SPA fallback: 非 API 路径一律返回 index.html，让前端路由处理。"
012    [request]
013    (if (and (string? (:uri request))
014             (not (str/starts-with? (:uri request) "/api/")))
015      (-> (response/resource-response "public/index.html")
016          (response/content-type "text/html; charset=utf-8"))
017      (-> {:status 404 :body "Not found"}
018          (response/content-type "text/plain"))))
019  
020  (defonce ^:private ring-handler-atom (atom nil))
021  
022  (defn ring-handler
023    "动态 Ring handler 入口。追踪功能可以通过更新 atom 来切换实际处理函数。"
024    [request]
025    (@ring-handler-atom request))
026  
027  (defn set-ring-handler!
028    "供调用追踪模块动态替换实际 handler。"
029    [f]
030    (reset! ring-handler-atom f))
031  
032  (defn current-ring-handler
033    "返回当前实际的 Ring handler（不是动态入口）。"
034    []
035    @ring-handler-atom)
036  
037  (defmethod ig/init-key :handler/ring
038    [_ {:keys [router api-path] :as opts}]
039    (let [actual (ring/ring-handler
040                  (router)
041                  (ring/routes
042                    ;; Handle trailing slash in routes - add it + redirect to it
043                    ;; https://github.com/metosin/reitit/blob/master/doc/ring/slash_handler.md
044                   (ring/redirect-trailing-slash-handler)
045                   (ring/create-resource-handler {:path "/"})
046                   (when (some? api-path)
047                     (swagger-ui/create-swagger-ui-handler {:path api-path
048                                                            :url  (str api-path "/swagger.json")}))
049                    ;; SPA fallback: 所有非 API 404 返回 index.html
050                   (ring/create-default-handler
051                    {:not-found spa-not-found-handler
052                     :method-not-allowed
053                     (constantly (-> {:status 405, :body "Not allowed"}
054                                    (response/content-type "text/plain")))
055                     :not-acceptable
056                     (constantly (-> {:status 406, :body "Not acceptable"}
057                                    (response/content-type "text/plain")))}))
058                  {:middleware [(middleware/wrap-base opts)]})]
059      (reset! ring-handler-atom actual)
060      ring-handler))
061  
062  (defmethod ig/init-key :router/routes
063    [_ {:keys [routes]}]
064    (mapv (fn [route]
065            (if (fn? route)
066              (route)
067              route))
068          routes))
069  
070  (defmethod ig/init-key :router/core
071    [_ {:keys [routes env] :as opts}]
072    (if (= env :dev)
073      #(ring/router ["" opts routes] {:reitit.router/sequential true :conflicts nil})
074      (constantly (ring/router ["" opts routes] {:reitit.router/sequential true :conflicts nil}))))
