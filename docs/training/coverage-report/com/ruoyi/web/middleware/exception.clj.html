001  (ns com.ruoyi.web.middleware.exception
002    (:require
003     [clojure.tools.logging :as log]
004     [cheshire.core :as json]
005     [reitit.ring.middleware.exception :as exception]))
006  
007  (defn handler [message status exception request]
008    (when (>= status 500)
009      (log/error exception "Exception:" (.getMessage exception)))
010    {:status  status
011     :headers {"content-type" "application/json;charset=utf-8"}
012     :body    (json/generate-string
013               {:message   message
014                :exception (.getName (.getClass exception))
015                :data      (ex-data exception)
016                :uri       (:uri request)})})
017  
018  (def wrap-exception
019    (exception/create-exception-middleware
020     (merge
021      exception/default-handlers
022      {:system.exception/internal     (partial handler "internal exception" 500)
023       :system.exception/business     (partial handler "bad request" 400)
024       :system.exception/not-found    (partial handler "not found" 404)
025       :system.exception/unauthorized (partial handler "unauthorized" 401)
026       :system.exception/forbidden    (partial handler "forbidden" 403)
027  
028         ;; override the default handler
029       ::exception/default            (partial handler "default" 500)
030  
031         ;; print stack-traces for all exceptions
032       ::exception/wrap               (fn [handler e request]
033                                        (handler e request))})))
