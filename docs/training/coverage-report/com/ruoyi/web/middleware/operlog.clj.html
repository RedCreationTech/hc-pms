001  (ns com.ruoyi.web.middleware.operlog
002    "操作日志中间件：自动记录所有 API 请求到 sys_oper_log 表。"
003    (:require
004     [clojure.tools.logging :as log]
005     [clojure.string :as str]
006     [cheshire.core :as json]
007     [ring.util.response :as response]))
008  
009  (def ^:private skip-paths
010    "不记录日志的路径"
011    #{"/api/auth/login" "/api/health" "/api/user/profile"})
012  
013  (def ^:private get-methods
014    "GET 请求不记录日志（仅 health 和 login 特殊处理）"
015    #{:get :head :options})
016  
017  (defn- format-params
018    "格式化请求参数，过长时截断。"
019    [params]
020    (let [s (if (instance? String params)
021              params
022              (try (json/generate-string params)
023                   (catch Exception _ (str params))))]
024      (if (> (count s) 200)
025        (str (subs s 0 200) "...")
026        s)))
027  
028  (defn wrap-oper-log
029    "操作日志中间件包装器。
030    调用 query-fn 写入 sys_oper_log 表。"
031    [handler]
032    (fn [request]
033      (let [uri (:uri request)
034            start (System/currentTimeMillis)
035            response (handler request)
036            cost-ms (- (System/currentTimeMillis) start)]
037        (when (and (str/starts-with? uri "/api/")
038                   (not (skip-paths uri))
039                   (not (contains? get-methods (:request-method request))))
040          (let [identity (:identity request)
041                log-entry {:title       (str (name (:request-method request)) " " uri)
042                           :business_type 0
043                           :method      ""
044                           :request_method (name (:request-method request))
045                           :operator_type 1
046                           :oper_name   (or (:user-name identity) "anonymous")
047                           :dept_name   ""
048                           :oper_url    uri
049                           :oper_ip     (get-in request [:headers "x-forwarded-for"]
050                                                (:remote-addr request "127.0.0.1"))
051                           :oper_location ""
052                           :oper_param  (format-params (:params request))
053                           :json_result (str (:status response))
054                           :status      (if (>= (:status response) 400) 1 0)
055                           :error_msg   ""
056                           :cost_time   cost-ms}]
057            (try
058              (when-let [query-fn (get-in request [:components :query-fn])]
059                (query-fn :create-oper-log! log-entry))
060              (catch Exception e
061                (log/warn e "Failed to write operation log")))))
062        response)))
