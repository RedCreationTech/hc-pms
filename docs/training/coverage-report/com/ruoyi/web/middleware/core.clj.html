001  (ns com.ruoyi.web.middleware.core
002    (:require
003     [cheshire.core :as json]
004     [com.ruoyi.env :as env]
005     [com.ruoyi.web.middleware.operlog :as operlog]
006     [ring.middleware.defaults :as defaults]
007     [ring.middleware.session.cookie :as cookie]))
008  
009  (defn- wrap-cors
010    "允许跨域请求，支持前端开发服务器访问。"
011    [handler]
012    (fn [request]
013      (let [response (handler request)]
014        (-> response
015            (assoc-in [:headers "Access-Control-Allow-Origin"] "*")
016            (assoc-in [:headers "Access-Control-Allow-Methods"] "GET, POST, PUT, DELETE, OPTIONS")
017            (assoc-in [:headers "Access-Control-Allow-Headers"] "Content-Type, Authorization")
018            (assoc-in [:headers "Access-Control-Allow-Credentials"] "true")))))
019  
020  (defn- handle-preflight
021    "处理 CORS 预检请求。"
022    [handler]
023    (fn [request]
024      (if (= :options (:request-method request))
025        {:status 200
026         :headers {"Access-Control-Allow-Origin" "*"
027                   "Access-Control-Allow-Methods" "GET, POST, PUT, DELETE, OPTIONS"
028                   "Access-Control-Allow-Headers" "Content-Type, Authorization"
029                   "Access-Control-Allow-Credentials" "true"}
030         :body ""}
031        (handler request))))
032  
033  (defn- wrap-query-fn
034    "将 query-fn 注入到请求的 :components 中，供 operlog 中间件使用。"
035    [handler query-fn]
036    (fn [request]
037      (handler (assoc request :components {:query-fn query-fn}))))
038  
039  (defn- wrap-json-body
040    "确保响应 body 是字符串（JSON），而非 Clojure 数据结构。"
041    [handler]
042    (fn [request]
043      (let [resp (handler request)]
044        (if (map? (:body resp))
045          (-> resp
046              (assoc :body (json/generate-string (:body resp)))
047              (assoc-in [:headers "content-type"] "application/json;charset=utf-8"))
048          resp))))
049  
050  (defn wrap-base
051    [{:keys [metrics site-defaults-config cookie-secret query-fn] :as opts}]
052    (let [cookie-store (cookie/cookie-store {:key (.getBytes ^String cookie-secret)})]
053      (fn [handler]
054        (-> ((:middleware env/defaults) handler opts)
055            (defaults/wrap-defaults
056             (assoc-in site-defaults-config [:session :store] cookie-store))
057            wrap-cors
058            handle-preflight
059            operlog/wrap-oper-log
060            (wrap-query-fn query-fn)
061            wrap-json-body))))
