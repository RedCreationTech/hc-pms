001  (ns com.ruoyi.web.middleware.auth
002    "认证与授权中间件，提供 JWT 校验、在线心跳和权限拦截。"
003    (:require
004     [com.ruoyi.infra.security :as security]
005     [com.ruoyi.infra.online :as online]
006     [ring.util.response :as response]))
007  
008  (defn wrap-jwt-auth
009    "为请求附加当前认证用户，并更新在线心跳。
010    如果令牌无效，继续执行但 :identity 为 nil。"
011    [handler]
012    (fn [request]
013      (let [token (security/extract-token request)
014            claims (when token (security/parse-token token))
015            blacklisted? (and token (online/blacklisted? token))
016            _ (when (and claims (not blacklisted?)) (online/heartbeat! token))
017            request (if (and claims (not blacklisted?))
018                      (assoc request :identity claims)
019                      request)]
020        (handler request))))
021  
022  (defn require-auth
023    "要求请求必须通过认证，否则返回 401。"
024    [handler]
025    (fn [request]
026      (if (:identity request)
027        (handler request)
028        (-> (response/response {:code 401 :msg "未登录或令牌已过期"})
029            (response/status 401)
030            (response/content-type "application/json")))))
031  
032  (defn require-perms
033    "要求当前用户拥有指定权限中的任意一个，否则返回 403。"
034    [perms]
035    (let [required (set (if (sequential? perms) perms [perms]))]
036      (fn [handler]
037        (fn [request]
038          (let [user-perms (set (get-in request [:identity :perms] []))]
039            (if (some required user-perms)
040              (handler request)
041              (-> (response/response {:code 403 :msg "没有操作权限"})
042                  (response/status 403)
043                  (response/content-type "application/json"))))))))
044  
045  (defn auth-middleware
046    "组合中间件：JWT 解析 + 在线心跳 + 可选认证要求。"
047    ([] (auth-middleware {}))
048    ([{:keys [required? perms]}]
049     (fn [handler]
050       (let [h (if required? (require-auth handler) handler)
051             h (if perms ((require-perms perms) h) h)
052             h (wrap-jwt-auth h)]
053         h))))
