001  (ns com.ruoyi.web.routes.gen
002    "代码生成器路由。"
003    (:require
004     [com.ruoyi.web.controllers.gen :as gen]
005     [com.ruoyi.web.middleware.auth :as auth-mw]))
006  
007  (defn gen-routes [{:keys [gen-service]}]
008    ["/tool"
009     {:middleware [(auth-mw/auth-middleware {:required? true})]
010      :swagger {:tags ["代码生成"]}}
011     ["/gen"
012      ["/tables" {:get {:summary    "查询数据库表"
013                        :description "查询系统所有数据库表（供选择生成）"
014                        :handler    (partial gen/list-tables {:gen-service gen-service})}}]
015      ["/columns" {:get {:summary    "查询表列信息"
016                         :description "查询指定表的字段元数据"
017                         :handler    (partial gen/table-columns {:gen-service gen-service})}}]
018      ["/preview" {:get {:summary    "预览代码"
019                         :description "预览生成的代码模板内容"
020                         :parameters {:query [:map [:tableName :string]]}
021                         :handler    (partial gen/preview-code {:gen-service gen-service})}}]
022      ["/generate" {:post {:summary    "批量生成代码"
023                           :description "选择表并生成完整 CRUD 代码文件"
024                           :handler    (partial gen/batch-generate {:gen-service gen-service})}}]
025      ["/deploy" {:post {:summary    "部署代码"
026                         :description "将生成的代码部署到项目目录"
027                         :handler    (partial gen/deploy-code {:gen-service gen-service})}}]
028      ["/download" {:post {:summary    "下载代码"
029                           :description "批量生成代码并打包成 ZIP 下载"
030                           :handler    (partial gen/download-code {:gen-service gen-service})}}]]])
