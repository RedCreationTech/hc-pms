001  (ns com.ruoyi.web.routes.auth
002    "认证路由。"
003    (:require
004     [com.ruoyi.web.controllers.auth :as auth]
005     [com.ruoyi.web.controllers.register :as register]
006     [com.ruoyi.web.middleware.auth :as auth-mw]
007     [com.ruoyi.infra.security :as security]
008     [ring.util.response :as response]))
009  
010  (defn- wrap-parse-token
011    "简单的 JWT 解析中间件，不依赖 auth-middleware 的复杂逻辑。"
012    [handler]
013    (fn [request]
014      (let [token (some-> (get-in request [:headers "authorization"])
015                          (clojure.string/replace-first #"(?i)Bearer\s+" ""))
016            claims (when token (security/parse-token token))
017            request (if claims (assoc request :identity claims) request)]
018        (handler request))))
019  
020  (defn- require-identity
021    [handler]
022    (fn [request]
023      (if (:identity request)
024        (handler request)
025        (-> (response/response {:code 401 :msg "未登录或令牌已过期"})
026            (response/status 401)
027            (response/content-type "application/json")))))
028  
029  (defn auth-routes [{:keys [user-service log-service menu-service]}]
030    ["/auth"
031     {:swagger {:tags ["认证"]}}
032     ["/login" {:post {:summary    "登录"
033                       :description "用户名密码登录，返回 Token"
034                       :handler    (partial auth/login {:user-service user-service :log-service log-service})}}]
035     ["/logout" {:post {:summary     "退出登录"
036                        :description "清除当前用户会话"
037                        :middleware  [wrap-parse-token require-identity]
038                        :handler     (partial auth/logout {})}}]
039     ["/getInfo" {:get {:summary     "获取用户信息"
040                        :description "获取当前用户信息（角色/权限/菜单）"
041                        :middleware  [wrap-parse-token require-identity]
042                        :handler     (partial auth/get-info {:user-service user-service :menu-service menu-service})}}]
043     ["/register" {:post {:summary "用户注册" :handler (partial register/register {:user-service user-service})}}]])
