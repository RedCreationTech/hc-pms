001  (ns com.ruoyi.web.controllers.captcha
002    "验证码控制器 — 生成图片验证码。"
003    (:require
004     [ring.util.response :as response])
005    (:import [java.awt Color Font RenderingHints]
006             [java.awt.image BufferedImage]
007             [javax.imageio ImageIO]
008             [java.io ByteArrayOutputStream]
009             [java.util Random]))
010  
011  ;; 验证码存储（实际项目应用 Redis）
012  (defonce captcha-store (atom {}))
013  
014  (defn- generate-code
015    "生成随机验证码。"
016    [length]
017    (let [chars "ABCDEFGHJKLMNPQRSTUVWXYZ23456789"
018          random (Random.)]
019      (apply str (repeatedly length #(nth chars (.nextInt random (count chars)))))))
020  
021  (defn- generate-color
022    "生成随机颜色。"
023    [min-val max-val]
024    (let [random (Random.)
025          r (+ min-val (.nextInt random (- max-val min-val)))
026          g (+ min-val (.nextInt random (- max-val min-val)))
027          b (+ min-val (.nextInt random (- max-val min-val)))]
028      (Color. r g b)))
029  
030  (defn- create-captcha-image
031    "创建验证码图片。"
032    [code width height]
033    (let [image (BufferedImage. width height BufferedImage/TYPE_INT_RGB)
034          g (.createGraphics image)
035          random (Random.)]
036      ;; 设置背景
037      (.setColor g Color/WHITE)
038      (.fillRect g 0 0 width height)
039  
040      ;; 设置字体
041      (.setFont g (Font. "Arial" Font/BOLD 36))
042      (.setRenderingHint g RenderingHints/KEY_ANTIALIASING RenderingHints/VALUE_ANTIALIAS_ON)
043  
044      ;; 绘制验证码字符
045      (dotimes [i (count code)]
046        (.setColor g (generate-color 50 180))
047        (.drawString g (str (nth code i)) (+ 15 (* i 40)) (+ 35 (.nextInt random 10))))
048  
049      ;; 绘制干扰线
050      (dotimes [_ 6]
051        (.setColor g (generate-color 100 200))
052        (.drawLine g (.nextInt random width) (.nextInt random height)
053                   (.nextInt random width) (.nextInt random height)))
054  
055      ;; 绘制干扰点
056      (dotimes [_ 30]
057        (.setColor g (generate-color 150 230))
058        (.drawOval g (.nextInt random width) (.nextInt random height) 2 2))
059  
060      (.dispose g)
061      image))
062  
063  (defn captcha-image
064    "生成验证码图片并返回。"
065    [_ request]
066    (let [code (generate-code 4)
067          image (create-captcha-image code 150 50)
068          baos (ByteArrayOutputStream.)]
069      (ImageIO/write image "png" baos)
070      (let [uuid (or (get-in request [:query-params "r"])
071                     (str (java.util.UUID/randomUUID)))]
072        ;; 存储验证码，5分钟有效
073        (swap! captcha-store assoc uuid {:code code :expire (+ (System/currentTimeMillis) 300000)})
074        ;; 清理过期验证码
075        (let [now (System/currentTimeMillis)]
076          (swap! captcha-store #(into {} (filter (fn [[_ v]] (< now (:expire v))) %))))
077        ;; 返回图片和 UUID
078        (-> (response/response (.toByteArray baos))
079            (response/content-type "image/png")
080            (response/header "Captcha-UUID" uuid)))))
