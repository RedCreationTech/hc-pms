001  (ns com.ruoyi.web.controllers.auth
002    "认证控制器，处理登录、登出及当前用户信息获取。"
003    (:require
004     [com.ruoyi.domain.system.user :as user-service]
005     [com.ruoyi.domain.system.role :as role-service]
006     [com.ruoyi.domain.system.menu :as menu-service]
007     [com.ruoyi.infra.security :as security]
008     [com.ruoyi.infra.online :as online]
009     [com.ruoyi.web.controllers.captcha :as captcha]
010     [com.ruoyi.domain.system.log :as log-domain]
011     [ring.util.response :as response]
012     [clojure.string :as str]))
013  
014  (defn- success
015    "构造成功响应。"
016    [data]
017    (-> (response/response {:code 200 :msg "操作成功" :data data})
018        (response/content-type "application/json")))
019  
020  (defn- error
021    "构造错误响应。"
022    [code msg]
023    (-> (response/response {:code code :msg msg})
024        (response/status (if (>= code 500) 500 200))
025        (response/content-type "application/json")))
026  
027  (defn login
028    "用户登录，验证密码后签发 JWT，并注册在线用户。"
029    [{:keys [user-service log-service]} request]
030    (let [{:keys [username password captcha uuid]} (:body-params request)
031          login-ip (get-in request [:headers "x-forwarded-for"] (:remote-addr request "127.0.0.1"))
032          ;; 验证码校验
033          captcha-valid? (if (and (seq uuid) (seq captcha))
034                           (let [stored (get @captcha/captcha-store uuid)]
035                             (and stored
036                                  (<= (System/currentTimeMillis) (:expire stored))
037                                  (= (.toUpperCase captcha) (.toUpperCase (:code stored)))))
038                           true)]
039      (when (seq uuid) (swap! captcha/captcha-store dissoc uuid))
040      (if (not captcha-valid?)
041        (error 400 "验证码错误或已过期")
042        (if (or (str/blank? username) (str/blank? password))
043          (error 400 "用户名和密码不能为空")
044          (if-let [user (user-service/find-user-by-name user-service username)]
045            (if (security/verify-password password (:password user))
046              (if (= "0" (:status user))
047                (let [roles (user-service/find-user-by-id user-service (:user_id user))
048                      role-ids (mapv :role_id (:roles roles))
049                      token (security/generate-token (:user_id user) (:user_name user) role-ids)
050                      _ (online/register! token (:user_name user) login-ip)
051                      ;; 记录登录日志
052                      _ (log-domain/create-login-log! user-service
053                                                      {:user_name username :ipaddr login-ip :login_location ""
054                                                       :browser "" :os "" :status "0" :msg "登录成功"})]
055                  (success {:token token}))
056                (error 403 "用户已被停用"))
057              (do
058                (log-domain/create-login-log! user-service
059                                              {:user_name username :ipaddr login-ip :login_location ""
060                                               :browser "" :os "" :status "1" :msg "密码错误"})
061                (error 400 "密码错误")))
062            (error 400 "用户不存在"))))))
063  
064  (defn get-info
065    "获取当前登录用户信息及权限菜单。"
066    [{:keys [user-service menu-service]} request]
067    (let [identity (:identity request)
068          user-id (:user-id identity)]
069      (if-let [user (user-service/find-user-by-id user-service user-id)]
070        (let [roles (:roles user)
071              role-ids (mapv :role_id roles)
072              perms (->> (mapcat #(role-service/get-role-perms {:query-fn (:query-fn user-service)} %) role-ids)
073                         (into #{})
074                         (vec))
075              menus (menu-service/menu-tree-by-roles menu-service role-ids)]
076          (success {:user (select-keys user [:user_id :user_name :nick_name :avatar :email :phonenumber :sex])
077                    :roles (mapv :role_key roles)
078                    :permissions perms
079                    :menus menus}))
080        (error 401 "用户不存在"))))
081  
082  (defn logout
083    "用户登出，清除在线记录。"
084    [request]
085    (when-let [token (security/extract-token request)]
086      (online/unregister! token))
087    (success {}))
