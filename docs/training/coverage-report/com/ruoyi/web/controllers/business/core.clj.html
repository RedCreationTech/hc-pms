290  (defn upload-attachment [{:keys [business-service]} request]
291    (try
292      (let [query-fn (:query-fn business-service)
293            file (get-in request [:params :file])
294            body (:params request)
295            temp-file (:tempfile file)
296            original (:filename file)
297            biz-type (or (:biz_type body) "common")
298            biz-id (parse-int* (:biz_id body) nil)
299            section-key (or (:section_key body) "")
300            purpose (or (:file_purpose body) "attachment")]
301        (if (and temp-file original)
302          (let [dir (str upload-root "/" biz-type)
303                stored (str (java.util.UUID/randomUUID) "-" original)
304                target (io/file dir stored)
305                ext (extension original)]
306            (ensure-dir! dir)
307            (io/copy temp-file target)
308            (let [params {:biz_type biz-type
309                          :biz_id biz-id
310                          :section_key section-key
311                          :file_purpose purpose
312                          :original_name original
313                          :stored_name stored
314                          :storage_type "local"
315                          :storage_path (.getPath target)
316                          :file_url (str "/api/business/attachments/" stored "/download")
317                          :mime_type (or (:content-type file) "application/octet-stream")
318                          :extension ext
319                          :file_size (.length target)
320                          :sort_order (parse-int* (:sort_order body) 0)
321                          :create_by (current-user request)
322                          :remark (or (:remark body) "")}]
323              (query-fn :create-attachment! params)
324              (ok (assoc params :id (last-id business-service)))))
325          (fail "上传失败")))
326      (catch Exception e (fail (.getMessage e)))))

337    (let [id-or-name (get-in request [:path-params :id])
338          row (if (re-matches #"\d+" (str id-or-name))
339                (q-fn business-service :get-attachment {:id (parse-int* id-or-name nil)})
340                nil)
341          file (when row (io/file (:storage_path row)))]
342      (if (and file (.exists file))
343        (-> (response/response file)
344            (response/header "Content-Disposition" (str "attachment; filename=\"" (:original_name row) "\""))

346        (fail "文件不存在"))))
