008  (defn- ok
009    ([data] (ok 200 "操作成功" data))
010    ([code msg data]
011     (-> (response/response {:code code :msg msg :data data})
012         (response/content-type "application/json"))))

014  (defn- parse-int [v]
015    (when v (Integer/parseInt v)))
016  
017  (defn- page-params [request]
018    (let [q (:query-params request)]
019      {:page-num (or (parse-int (get q "page")) 1)
020       :page-size (or (parse-int (get q "size")) 10)
021       :user_name (get q "user_name")
022       :phonenumber (get q "phonenumber")
023       :status (get q "status")
024       :dept_id (get q "dept_id")}))
025  
026  (defn- fail [msg]
027    (-> (response/response {:code 500 :msg msg})
028        (response/content-type "application/json")))

030  (defn list-users
031    "查询用户列表（带数据权限过滤）。"
032    [{:keys [user-service]} request]
033    (let [raw (:query-params request)
034          identity (:identity request)
035          data-perm-filter (data-perm/data-perm-filter identity "default" :alias "u")
036          params (merge {:page-num (or (parse-int (get raw "page")) 1)
037                         :page-size (or (parse-int (get raw "size")) 10)}
038                        (dissoc raw "page" "size")
039                        (:params data-perm-filter))
040          result (user-service/list-users user-service params)]
041      (ok {:total (:total result) :rows (:rows result)})))

043  (defn get-user
044    "获取用户详情。"
045    [{:keys [user-service]} request]
046    (let [user-id (parse-long (get-in request [:path-params :id]))]
047      (if-let [user (user-service/find-user-by-id user-service user-id)]
048        (ok user)
049        (fail "用户不存在"))))

051  (defn create-user
052    "创建用户。"
053    [{:keys [user-service]} request]
054    (try
055      (let [body (:body-params request)
056            identity (:identity request)
057            params (merge {:dept_id 1 :user_type "00" :sex "0" :status "0"
058                           :email "" :phonenumber "" :avatar "" :remark ""
059                           :create_by (:user_name identity "")
060                           :roles [] :posts []}
061                          body)]
062        (when-not (:user_name params)
063          (throw (Exception. "用户名不能为空")))
064        (when-not (:nick_name params)
065          (throw (Exception. "用户昵称不能为空")))
066        (when-not (:password params)
067          (throw (Exception. "密码不能为空")))
068        (let [user-id (user-service/create-user! user-service params)]
069          (ok (str "创建成功: " user-id))))
070      (catch Exception e
071        (fail (.getMessage e)))))
072  
073  (defn update-user
074    "更新用户。"
075    [{:keys [user-service]} request]
076    (try
077      (let [user-id (parse-long (get-in request [:path-params :id]))
078            body (:body-params request)
079            identity (:identity request)
080            existing (user-service/find-user-by-id user-service user-id)
081            params (merge (select-keys existing [:dept_id :user_name :nick_name :user_type
082                                                 :email :phonenumber :sex :avatar
083                                                 :status :remark :password])
084                          (dissoc body :roles :posts)
085                          {:user-id user-id :update_by (:user_name identity "")}
086                          (when (:roles body) {:roles (:roles body)})
087                          (when (:posts body) {:posts (:posts body)}))]
088        (user-service/update-user! user-service params)
089        (ok "更新成功"))
090      (catch Exception e
091        (fail (.getMessage e)))))

134  (defn import-users
135    "导入用户。"
136    [{:keys [user-service]} request]
137    (try
138      (let [body (:body-params request)
139            rows (:rows body)
140            identity (:identity request)
141            results (mapv (fn [row]
142                            (try
143                              (let [params (merge {:dept_id 1 :user_type "00" :sex "0" :status "0"
144                                                   :email "" :phonenumber "" :avatar "" :remark ""
145                                                   :create_by (:user_name identity "") :password "123456"
146                                                   :roles [] :posts []}
147                                                  row)]
148                                (when (:user_name params)
149                                  (user-service/create-user! user-service params))
150                                {:user_name (:user_name params) :status "success"})
151                              (catch Exception e
152                                {:user_name (:user_name row) :status "failed" :msg (.getMessage e)})))
153                          rows)]
154        (ok {:total (count results)
155             :success (count (filter #(= "success" (:status %)) results))
156             :failed (count (filter #(= "failed" (:status %)) results))
157             :details results}))
158      (catch Exception e
159        (fail (.getMessage e)))))

163    [{:keys [user-service]} request]
164    (try
165      (let [identity (:identity request)
166            raw (:query-params request)

173            csv-header "user_name,nick_name,email,phonenumber,sex,status,dept_id,remark"
174            csv-rows (mapv (fn [r]

176                                  (:phonenumber r) "," (:sex r) "," (:status r) ","
177                                  (:dept_id r) "," (:remark r)))
178                           rows)
179            csv (str csv-header "\n" (clojure.string/join "\n" csv-rows))]
180        (-> (response/response csv)
181            (response/header "Content-Type" "text/csv; charset=utf-8")
182            (response/header "Content-Disposition" "attachment; filename=users.csv")))
183      (catch Exception e
184        (fail (.getMessage e)))))
