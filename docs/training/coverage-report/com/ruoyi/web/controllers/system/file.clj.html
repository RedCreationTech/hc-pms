001  (ns com.ruoyi.web.controllers.system.file
002    "文件管理控制器 — 上传、列表、下载、删除。"
003    (:require
004     [clojure.java.io :as io]
005     [ring.util.response :as response]))
006  
007  (def upload-dir "uploads/")
008  
009  (defn- ensure-dir! []
010    (let [dir (io/file upload-dir)]
011      (when-not (.exists dir) (.mkdirs dir))))
012  
013  (defn- ok
014    ([data] (ok 200 "操作成功" data))
015    ([code msg data]
016     (-> (response/response {:code code :msg msg :data data})
017         (response/content-type "application/json"))))
018  
019  (defn list-files
020    "获取上传文件列表。"
021    [_ _]
022    (ensure-dir!)
023    (let [dir (io/file upload-dir)
024          files (when (.exists dir)
025                  (->> (.listFiles dir)
026                       (filter #(.isFile %))
027                       (mapv (fn [f]
028                               {:name (.getName f)
029                                :size (.length f)
030                                :modified (.lastModified f)}))
031                       (sort-by :modified)
032                       reverse))]
033      (ok files)))
034  
035  (defn upload-file
036    "上传文件。"
037    [_ request]
038    (try
039      (ensure-dir!)
040      (let [file (get-in request [:params :file])
041            temp-file (:tempfile file)
042            filename (:filename file)]
043        (if (and temp-file filename)
044          (let [target (io/file upload-dir filename)]
045            (io/copy temp-file target)
046            (ok {:name filename :size (.length target)}))
047          (ok 500 "上传失败" nil)))
048      (catch Exception e
049        (ok 500 (.getMessage e) nil))))
050  
051  (defn download-file
052    "下载文件。"
053    [_ request]
054    (let [filename (get-in request [:path-params :filename])
055          file (io/file upload-dir filename)]
056      (if (.exists file)
057        (-> (response/response file)
058            (response/header "Content-Disposition" (str "attachment; filename=\"" filename "\""))
059            (response/content-type "application/octet-stream"))
060        (ok 404 "文件不存在" nil))))
061  
062  (defn delete-file
063    "删除文件。"
064    [_ request]
065    (let [filename (get-in request [:path-params :filename])
066          file (io/file upload-dir filename)]
067      (if (.exists file)
068        (do (.delete file) (ok "删除成功"))
069        (ok 404 "文件不存在" nil))))
