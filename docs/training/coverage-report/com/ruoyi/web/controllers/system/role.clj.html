007  (defn- ok ([data] (ok 200 "操作成功" data))
008    ([code msg data]
009     (-> (response/response {:code code :msg msg :data data})
010         (response/content-type "application/json"))))
011  
012  (defn- fail [msg]
013    (-> (response/response {:code 500 :msg msg})
014        (response/content-type "application/json")))
015  
016  (defn list-roles
017    "查询角色列表。"
018    [{:keys [role-service]} request]
019    (let [params (:query-params request)]
020      (ok (role-service/list-roles role-service params))))
021  
022  (defn get-role
023    "查询角色详情。"
024    [{:keys [role-service]} request]
025    (let [role-id (parse-long (get-in request [:path-params :id]))]
026      (if-let [role (role-service/find-role-by-id role-service role-id)]
027        (ok role)
028        (fail "角色不存在"))))
029  
030  (defn create-role
031    "新增角色。"
032    [{:keys [role-service]} request]
033    (try
034      (let [role-id (role-service/create-role! role-service (:body-params request))]
035        (ok (str "创建成功: " role-id)))
036      (catch Exception e (fail (.getMessage e)))))
037

063  (defn- parse-user-ids
064    "将逗号分隔的用户ID字符串解析为long集合。"
065    [s]
066    (when (seq s)
067      (->> (clojure.string/split s #",")
068           (map clojure.string/trim)
069           (remove empty?)
070           (map parse-long)
071           (doall))))
