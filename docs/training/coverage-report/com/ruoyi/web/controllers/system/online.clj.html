001  (ns com.ruoyi.web.controllers.system.online
002    "在线用户控制器。"
003    (:require
004     [ring.util.response :as response]))
005  
006  (defn- ok [data]
007    (-> (response/response {:code 200 :msg "操作成功" :data data})
008        (response/content-type "application/json")))
009  
010  (defn- success [msg]
011    (-> (response/response {:code 200 :msg msg})
012        (response/content-type "application/json")))
013  
014  (defn- parse-int [v]
015    (when v (Integer/parseInt v)))
016  
017  (defn list-online
018    "获取在线用户列表。"
019    [{:keys [online-service]} request]
020    (let [params (:query-params request)
021          page (or (parse-int (get params "pageNum")) 1)
022          size (or (parse-int (get params "pageSize")) 10)
023          result ((:list-online online-service)
024                  {:login-name (get params "user_name")
025                   :ipaddr (get params "ipaddr")
026                   :page-num page
027                   :page-size size})]
028      (ok {:rows (:rows result) :total (:total result)})))
029  
030  (defn force-logout
031    "强退指定用户。"
032    [{:keys [online-service]} request]
033    (let [token-id (get-in request [:path-params :token-id])]
034      ((:force-logout online-service) token-id)
035      (success "操作成功")))
