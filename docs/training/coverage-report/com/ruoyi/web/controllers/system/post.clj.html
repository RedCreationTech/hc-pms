001  (ns com.ruoyi.web.controllers.system.post
002    "岗位管理控制器。"
003    (:require
004     [com.ruoyi.domain.system.post :as post-service]
005     [ring.util.response :as response]))
006  
007  (defn- ok ([data] (ok 200 "操作成功" data))
008    ([code msg data]
009     (-> (response/response {:code code :msg msg :data data})
010         (response/content-type "application/json"))))
011  
012  (defn- fail [msg]
013    (-> (response/response {:code 500 :msg msg})
014        (response/content-type "application/json")))
015  
016  (defn- current-user-name [request]
017    (get-in request [:identity :user-name] ""))
018  
019  (defn list-posts
020    "查询岗位列表。"
021    [{:keys [post-service]} request]
022    (let [params (:query-params request)]
023      (ok (post-service/list-posts post-service params))))
024  
025  (defn get-post
026    [{:keys [post-service]} request]
027    (let [post-id (parse-long (get-in request [:path-params :id]))]
028      (if-let [post (post-service/find-post-by-id post-service post-id)]
029        (ok post)
030        (fail "岗位不存在"))))
031  
032  (defn create-post
033    [{:keys [post-service]} request]
034    (try
035      (let [params (assoc (:body-params request) :create_by (current-user-name request))
036            post-id (post-service/create-post! post-service params)]
037        (ok (str "创建成功: " post-id)))
038      (catch Exception e (fail (.getMessage e)))))
039  
040  (defn update-post
041    [{:keys [post-service]} request]
042    (try
043      (let [post-id (parse-long (get-in request [:path-params :id]))
044            params (-> (:body-params request)
045                       (assoc :post_id post-id)
046                       (assoc :update_by (current-user-name request)))]
047        (post-service/update-post! post-service params)
048        (ok "更新成功"))
049      (catch Exception e (fail (.getMessage e)))))
050  
051  (defn delete-post
052    [{:keys [post-service]} request]
053    (let [post-id (parse-long (get-in request [:path-params :id]))]
054      (post-service/delete-post! post-service post-id)
055      (ok "删除成功")))
056  
057  (defn change-status
058    "修改岗位状态。"
059    [{:keys [post-service]} request]
060    (let [post-id (parse-long (get-in request [:path-params :id]))
061          status (get-in request [:body-params :status])]
062      (post-service/update-post! post-service {:post_id post-id :status status :update_by (current-user-name request)})
063      (ok "状态修改成功")))
