001  (ns com.ruoyi.web.controllers.system.dict
002    "字典管理控制器。"
003    (:require
004     [com.ruoyi.domain.system.dict :as dict-service]
005     [com.ruoyi.infra.data-perm :as data-perm]
006     [ring.util.response :as response]))
007  
008  (defn- ok ([data] (ok 200 "操作成功" data))
009    ([code msg data]
010     (-> (response/response {:code code :msg msg :data data})
011         (response/content-type "application/json"))))
012  
013  (defn- fail [msg]
014    (-> (response/response {:code 500 :msg msg})
015        (response/content-type "application/json")))
016  
017  (defn- current-user-name [request]
018    (get-in request [:identity :user-name] ""))
019  
020  (defn list-dict-types
021    "查询字典类型列表（带数据权限过滤）。"
022    [{:keys [dict-service]} request]
023    (let [params (:query-params request)
024          identity (:identity request)
025          data-perm-filter (data-perm/data-perm-filter identity "default" :alias "u")
026          params (merge params (:params data-perm-filter))]
027      (ok (dict-service/list-dict-types dict-service params))))
028  
029  (defn get-dict-type
030    [{:keys [dict-service]} request]
031    (let [dict-id (parse-long (get-in request [:path-params :id]))]
032      (if-let [dt (dict-service/find-dict-type-by-id dict-service dict-id)]
033        (ok dt)
034        (fail "字典类型不存在"))))
035  
036  (defn create-dict-type
037    [{:keys [dict-service]} request]
038    (try
039      (let [params (assoc (:body-params request) :create_by (current-user-name request))
040            dict-id (dict-service/create-dict-type! dict-service params)]
041        (ok (str "创建成功: " dict-id)))
042      (catch Exception e (fail (.getMessage e)))))
043  
044  (defn update-dict-type
045    [{:keys [dict-service]} request]
046    (try
047      (let [dict-id (parse-long (get-in request [:path-params :id]))
048            params (-> (:body-params request)
049                       (assoc :dict_id dict-id)
050                       (assoc :update_by (current-user-name request)))]
051        (dict-service/update-dict-type! dict-service params)
052        (ok "更新成功"))
053      (catch Exception e (fail (.getMessage e)))))
054  
055  (defn delete-dict-type
056    [{:keys [dict-service]} request]
057    (let [dict-id (parse-long (get-in request [:path-params :id]))]
058      (dict-service/delete-dict-type! dict-service dict-id)
059      (ok "删除成功")))
060
