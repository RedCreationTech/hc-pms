001  (ns com.ruoyi.web.controllers.system.dept
002    "部门管理控制器。"
003    (:require
004     [com.ruoyi.domain.system.dept :as dept-service]
005     [ring.util.response :as response]))
006  
007  (defn- ok ([data] (ok 200 "操作成功" data))
008    ([code msg data]
009     (-> (response/response {:code code :msg msg :data data})
010         (response/content-type "application/json"))))
011  
012  (defn- fail [msg]
013    (-> (response/response {:code 500 :msg msg})
014        (response/content-type "application/json")))
015  
016  (defn- current-user-name [request]
017    (get-in request [:identity :user-name] ""))
018  
019  (defn list-depts
020    "查询部门列表。"
021    [{:keys [dept-service]} request]
022    (let [params (:query-params request)]
023      (ok (dept-service/list-depts dept-service params))))
024  
025  (defn dept-tree
026    "获取部门树（用于用户管理左侧选择）。"
027    [{:keys [dept-service]} _request]
028    (ok (dept-service/list-depts dept-service {})))
029  
030  (defn get-dept
031    [{:keys [dept-service]} request]
032    (let [dept-id (parse-long (get-in request [:path-params :id]))]
033      (if-let [dept (dept-service/find-dept-by-id dept-service dept-id)]
034        (ok dept)
035        (fail "部门不存在"))))
036  
037  (defn create-dept
038    [{:keys [dept-service]} request]
039    (try
040      (let [params (assoc (:body-params request) :create_by (current-user-name request))
041            dept-id (dept-service/create-dept! dept-service params)]
042        (ok (str "创建成功: " dept-id)))
043      (catch Exception e (fail (.getMessage e)))))
044  
045  (defn update-dept
046    [{:keys [dept-service]} request]
047    (try
048      (let [dept-id (parse-long (get-in request [:path-params :id]))
049            params (-> (:body-params request)
050                       (assoc :dept_id dept-id)
051                       (assoc :update_by (current-user-name request)))]
052        (dept-service/update-dept! dept-service params)
053        (ok "更新成功"))
054      (catch Exception e (fail (.getMessage e)))))
055  
056  (defn delete-dept
057    [{:keys [dept-service]} request]
058    (let [dept-id (parse-long (get-in request [:path-params :id]))]
059      (dept-service/delete-dept! dept-service dept-id)
060      (ok "删除成功")))
061  
062  (defn change-status
063    "修改部门状态。"
064    [{:keys [dept-service]} request]
065    (let [dept-id (parse-long (get-in request [:path-params :id]))
066          status (get-in request [:body-params :status])]
067      (dept-service/update-dept! dept-service {:dept_id dept-id :status status :update_by (current-user-name request)})
068      (ok "状态修改成功")))
