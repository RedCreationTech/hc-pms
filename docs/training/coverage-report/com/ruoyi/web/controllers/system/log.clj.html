001  (ns com.ruoyi.web.controllers.system.log
002    "日志审计控制器。"
003    (:require
004     [com.ruoyi.domain.system.log :as log-service]
005     [ring.util.response :as response]))
006  
007  (defn- ok ([data] (ok 200 "操作成功" data))
008    ([code msg data]
009     (-> (response/response {:code code :msg msg :data data})
010         (response/content-type "application/json"))))
011  
012  (defn list-oper-logs
013    "查询操作日志列表。"
014    [{:keys [log-service]} request]
015    (let [result (log-service/list-oper-logs log-service (:query-params request))]
016      (ok {:total (:total result) :rows (:rows result)})))
017  
018  (defn clear-oper-logs
019    "清空操作日志。"
020    [{:keys [log-service]} request]
021    (log-service/clear-oper-logs! log-service (:query-params request))
022    (ok "清空成功"))
023  
024  (defn list-login-logs

029  
030  (defn clear-login-logs

040      (ok {:total (:total result) :rows (:rows result)})))
041  
042  (defn kick-online-user
043    "强退在线用户。"
044    [{:keys [log-service]} request]
045    (let [session-id (get-in request [:path-params :id])]
046      (log-service/delete-online-user! log-service session-id)
047      (ok "强退成功")))
