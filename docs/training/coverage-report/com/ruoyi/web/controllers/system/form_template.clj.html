001  (ns com.ruoyi.web.controllers.system.form-template
002    "表单模板控制器。"
003    (:require
004     [clojure.string :as str]
005     [com.ruoyi.domain.system.form-template :as form-template-service]
006     [ring.util.response :as response]))
007  
008  (defn- ok
009    ([data] (ok 200 "操作成功" data))
010    ([code msg data]
011     (-> (response/response {:code code :msg msg :data data})
012         (response/content-type "application/json"))))
013  
014  (defn- fail [msg]
015    (-> (response/response {:code 500 :msg msg})
016        (response/content-type "application/json")))
017  
018  (defn- current-user-name [request]
019    (get-in request [:identity :user-name] ""))
020  
021  (defn- ->snake [params]
022    "将查询参数键统一转为 snake_case。"
023    (reduce-kv (fn [m k v]
024                 (assoc m
025                        (keyword (str/replace (name k) #"([a-z])([A-Z])" "$1_$2"))
026                        v))
027               {}
028               params))
029  
030  (defn list-form-templates
031    "查询表单模板列表。"
032    [{:keys [form-template-service]} request]
033    (let [params (->snake (:query-params request))]
034      (ok (form-template-service/list-form-templates form-template-service params))))
035  
036  (defn get-form-template
037    "根据ID获取表单模板。"
038    [{:keys [form-template-service]} request]
039    (let [id (parse-long (get-in request [:path-params :id]))]
040      (if-let [template (form-template-service/find-form-template-by-id form-template-service id)]
041        (ok template)
042        (fail "模板不存在"))))
043  
044  (defn create-form-template
045    "创建表单模板。"
046    [{:keys [form-template-service]} request]
047    (try
048      (let [params (assoc (:body-params request) :create_by (current-user-name request))
049            id (form-template-service/create-form-template! form-template-service params)]
050        (ok (str "创建成功: " id)))
051      (catch Exception e (fail (.getMessage e)))))
052  
053  (defn update-form-template
054    "更新表单模板。"
055    [{:keys [form-template-service]} request]
056    (try
057      (let [id (parse-long (get-in request [:path-params :id]))
058            params (-> (:body-params request)
059                       (assoc :id id)
060                       (assoc :update_by (current-user-name request)))]
061        (form-template-service/update-form-template! form-template-service params)
062        (ok "更新成功"))
063      (catch Exception e (fail (.getMessage e)))))
064  
065  (defn delete-form-template
066    "删除表单模板。"
067    [{:keys [form-template-service]} request]
068    (let [id (parse-long (get-in request [:path-params :id]))]
069      (form-template-service/delete-form-template! form-template-service id)
070      (ok "删除成功")))
