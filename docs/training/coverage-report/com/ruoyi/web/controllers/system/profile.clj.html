001  (ns com.ruoyi.web.controllers.system.profile
002    "个人中心控制器。"
003    (:require
004     [ring.util.response :as response]
005     [com.ruoyi.domain.system.user :as user-service]
006     [com.ruoyi.infra.security :as security]
007     [com.ruoyi.infra.online :as online]
008     [clojure.string :as str]))
009  
010  (defn- ok
011    ([data] (ok 200 "操作成功" data))
012    ([code msg data]
013     (-> (response/response {:code code :msg msg :data data})
014         (response/content-type "application/json"))))
015  
016  (defn- fail [msg]
017    (-> (response/response {:code 500 :msg msg})
018        (response/content-type "application/json")))
019  
020  (defn get-profile
021    "获取当前用户个人信息。"
022    [{:keys [user-service]} request]
023    (let [identity (:identity request)
024          user-id (:user-id identity)]
025      (if-let [user (user-service/find-user-by-id user-service user-id)]
026        (ok (select-keys user [:user_id :user_name :nick_name :avatar :email :phonenumber :sex]))
027        (fail "用户不存在"))))
028  
029  (defn update-profile
030    "更新当前用户个人信息。"
031    [{:keys [user-service]} request]
032    (try
033      (let [identity (:identity request)
034            user-id (:user-id identity)
035            params (assoc (:body-params request) :user-id user-id)]
036        (user-service/update-user! user-service params)
037        (ok "更新成功"))
038      (catch Exception e
039        (fail (.getMessage e)))))
040  
041  (defn- save-avatar!
042    "保存上传的头像文件。"
043    [upload]
044    (let [filename (str (System/currentTimeMillis) "_" (:filename upload))
045          upload-dir (or (System/getProperty "app.upload.dir") "uploads/avatar")
046          file (java.io.File. (str upload-dir "/" filename))]
047      (.mkdirs (.getParentFile file))
048      (clojure.java.io/copy (:tempfile upload) file)
049      (str "/" upload-dir "/" filename)))
050  
051  (defn upload-avatar
052    "上传头像。"
053    [{:keys [user-service]} request]
054    (try
055      (let [identity (:identity request)
056            user-id (:user-id identity)
057            {:keys [avatarfile]} (:params-params request)
058            avatar-url (when avatarfile (save-avatar! avatarfile))]
059        (when avatar-url
060          (user-service/update-user! user-service {:user-id user-id :avatar avatar-url}))
061        (ok {:avatar avatar-url}))
062      (catch Exception e
063        (fail (.getMessage e)))))
064  
065  (defn change-password
066    "修改当前用户密码。"
067    [{:keys [user-service]} request]
068    (try
069      (let [identity (:identity request)
070            user-id (:user-id identity)
071            {:keys [old_password new_password]} (:body-params request)]
072        (if (or (str/blank? old_password) (str/blank? new_password))
073          (fail "旧密码和新密码不能为空")
074          (let [user (user-service/find-user-by-id user-service user-id)]
075            (if (security/verify-password old_password (:password user))
076              (do
077                (user-service/update-user! user-service {:user-id user-id :password new_password})
078                (ok "密码修改成功"))
079              (fail "旧密码错误")))))
080      (catch Exception e
081        (fail (.getMessage e)))))
