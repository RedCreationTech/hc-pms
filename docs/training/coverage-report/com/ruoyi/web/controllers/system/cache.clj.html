015    {"user"

022     "notice"
023     {"1" {:notice_id 1 :notice_title "系统公告" :notice_type "1"}}})

025  (defonce cache-data
026    (atom (seed-cache)))
027  
028  (defn- ok
029    ([data] (ok 200 "操作成功" data))
030    ([code msg data]
031     (-> (response/response {:code code :msg msg :data data})
032         (response/content-type "application/json"))))
033  
034  (defn- total-memory-bytes []
035    (try
036      (let [rt (Runtime/getRuntime)]
037        (- (.totalMemory rt) (.freeMemory rt)))
038      (catch Exception _ 0)))
039  
040  (defn- memory-used-mb []
041    (quot (total-memory-bytes) 1048576))
042  
043  (defn- memory-max-mb []
044    (quot (.maxMemory (Runtime/getRuntime)) 1048576))
045  
046  (defn- total-keys []
047    (reduce + (map count (vals @cache-data))))
048  
049  (defn- command-stats []
050    (let [s @cache-stats]
051      [{:name "get" :value (:get s)}
052       {:name "hit" :value (:hit s)}
053       {:name "miss" :value (:miss s)}
054       {:name "clear" :value (:clear s)}]))

058  (defn cache-info
059    "获取缓存整体信息。"
060    [_ _]

062         :type "Clojure Atom"

069                       :keysCount (total-keys)

072  
073  (defn cache-names
074    "获取缓存名称列表。"
075    [_ _]
076    (ok {:cacheNames (vec (keys @cache-data))}))
077  
078  (defn cache-keys
079    "获取所有缓存键（扁平化）。"
080    [_ _]
081    (let [ks (for [[cache-name entries] @cache-data
082                   k (keys entries)]
083               (str cache-name "::" (name k)))]
084      (ok {:keys (vec ks)
085           :count (count ks)})))
086  
087  (defn cache-keys-by-name
088    "获取指定缓存名称下的键列表。"
089    [_ request]
090    (let [cache-name (get-in request [:path-params :cacheName])
091          entries (get @cache-data cache-name {})]
092      (ok {:cacheName cache-name
093           :keys (mapv name (keys entries))
094           :count (count entries)})))
095  
096  (defn cache-value
097    "获取缓存值。"
098    [_ request]
099    (let [cache-name (get-in request [:path-params :cacheName])
100          cache-key (get-in request [:path-params :cacheKey])
101          val (get-in @cache-data [cache-name cache-key])]
102      (swap! cache-stats update :get inc)
103      (if (some? val)
104        (swap! cache-stats update :hit inc)
105        (swap! cache-stats update :miss inc))
106      (ok {:cacheName cache-name
107           :cacheKey cache-key
108           :value (if (some? val) (json/write-str val) "")})))
109  
110  ;; ─── 清除操作 ──────────────────────────────────────────────────────
111  
112  (defn clear-cache
113    "清空所有缓存。"
114    [_ _]
115    (reset! cache-data (seed-cache))
116    (swap! cache-stats update :clear inc)
117    (ok "缓存已清空"))
118  
119  (defn clear-cache-name
120    "清除指定名称的缓存。"
121    [_ request]
122    (let [cache-name (get-in request [:path-params :cacheName])]
123      (swap! cache-data assoc cache-name {})
124      (swap! cache-stats update :clear inc)
125      (ok "缓存已清空")))
126  
127  (defn clear-cache-key
128    "清除指定键。"
129    [_ request]
130    (let [cache-name (get-in request [:path-params :cacheName])
131          cache-key (get-in request [:path-params :cacheKey])]
132      (swap! cache-data update cache-name dissoc cache-key)
133      (swap! cache-stats update :clear inc)
134      (ok "缓存键已清除")))
135  
136  (defn clear-cache-all
137    "清除所有缓存并重置统计。"
138    [_ _]
139    (reset! cache-data (seed-cache))
140    (reset! cache-stats {:get 0 :hit 0 :miss 0 :clear 0})
141    (ok "所有缓存已清空"))
