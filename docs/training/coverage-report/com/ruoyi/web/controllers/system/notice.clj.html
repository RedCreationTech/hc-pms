001  (ns com.ruoyi.web.controllers.system.notice
002    "通知公告控制器。"
003    (:require
004     [ring.util.response :as response]))
005  
006  (defn- ok
007    "构造成功响应。"
008    ([data] (ok 200 "操作成功" data))
009    ([code msg data]
010     (-> (response/response {:code code :msg msg :data data})
011         (response/content-type "application/json"))))
012  
013  (defn- fail
014    "构造失败响应。"
015    [msg]
016    (-> (response/response {:code 500 :msg msg})
017        (response/content-type "application/json")))
018  
019  (defn- parse-int
020    "将字符串解析为整数。"
021    [v]
022    (when v (Integer/parseInt v)))
023  
024  (defn list-notices
025    "查询通知公告列表。"
026    [{:keys [query-fn]} request]
027    (let [params (:query-params request)
028          page (or (parse-int (get params "page")) 1)
029          size (or (parse-int (get params "size")) 10)
030          offset (* (dec page) size)
031          query-params {:notice_name (get params "notice_name")
032                        :notice_type (get params "notice_type")
033                        :create_by   (get params "create_by")
034                        :page_size   size
035                        :offset      offset}
036          rows (query-fn :list-notices query-params)
037          total (query-fn :count-notices query-params)]
038      (ok {:rows rows :total (:total total)})))
039  
040  (defn get-notice
041    "获取通知公告详情。"
042    [{:keys [query-fn]} request]
043    (let [notice-id (parse-int (get-in request [:path-params :id]))]
044      (if-let [notice (query-fn :find-notice-by-id {:notice_id notice-id} {:result-set-fn first})]
045        (ok notice)
046        (fail "通知公告不存在"))))
047  
048  (defn create-notice
049    "新增通知公告。"
050    [{:keys [query-fn]} request]
051    (try
052      (let [body (:body-params request)
053            identity (:identity request)
054            params {:notice_name (:notice_name body)
055                    :notice_type (:notice_type body "1")
056                    :status      (:status body "0")
057                    :create_by   (:user-name identity "")
058                    :notice_content (:notice_content body "")
059                    :remark      (:remark body "")}]
060        (query-fn :create-notice! params)
061        (ok "创建成功"))
062      (catch Exception e (fail (.getMessage e)))))
063  
064  (defn update-notice
065    "更新通知公告。"
066    [{:keys [query-fn]} request]
067    (try
068      (let [notice-id (parse-int (get-in request [:path-params :id]))
069            body (:body-params request)
070            params {:notice_id   notice-id
071                    :notice_name (:notice_name body)
072                    :notice_type (:notice_type body)
073                    :status      (:status body)
074                    :notice_content (:notice_content body)
075                    :update_by   (get-in request [:identity :user-name] "")
076                    :remark      (:remark body)}]
077        (query-fn :update-notice! params)
078        (ok "更新成功"))
079      (catch Exception e (fail (.getMessage e)))))
080  
081  (defn delete-notice
082    "删除通知公告。"
083    [{:keys [query-fn]} request]
084    (let [notice-id (parse-int (get-in request [:path-params :id]))]
085      (query-fn :delete-notice! {:notice_id notice-id})
086      (ok 200 "删除成功" {})))
