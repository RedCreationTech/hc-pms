019    ([data] (ok 200 "操作成功" data))
020    ([code msg data]

028  (defn- parse-int [v]
029    (when (and v (not (str/blank? (str v))))
030      (try (Integer/parseInt (str v))
031           (catch Exception _ nil))))

033  (defn- csv-row->user
034    "将 CSV 行向量转换为用户参数映射。"
035    [headers row]
036    (let [m (zipmap headers row)]
037      {:user_name (str/trim (get m "user_name" ""))
038       :nick_name (str/trim (get m "nick_name" ""))
039       :email (str/trim (get m "email" ""))
040       :phonenumber (str/trim (get m "phonenumber" ""))
041       :sex (or (str/trim (get m "sex" "0")) "0")
042       :status (or (str/trim (get m "status" "0")) "0")
043       :dept_id (or (parse-int (get m "dept_id")) 1)
044       :user_type "00"
045       :avatar ""
046       :remark (str/trim (get m "remark" ""))}))

048  (defn- read-csv-rows
049    "读取 multipart 上传的 CSV 文件，返回行向量列表。"
050    [file]
051    (let [tempfile (:tempfile file)]
052      (with-open [reader (io/reader tempfile :encoding "UTF-8")]
053        (doall (csv/read-csv reader)))))

055  (defn import-users
056    "批量导入用户（multipart CSV）。"
057    [{:keys [user-service]} request]
058    (try
059      (let [multipart-params (:multipart-params request)
060            file (get multipart-params "file")
061            identity (:identity request)
062            _ (when (or (nil? file) (str/blank? (:filename file "")))
063                (throw (Exception. "请选择要上传的文件")))
064            rows (read-csv-rows file)
065            headers (mapv str/trim (first rows))
066            data-rows (rest rows)
067            default-password "123456"
068            results (mapv (fn [row]
069                            (try
070                              (let [user (csv-row->user headers row)]
071                                (when (str/blank? (:user_name user))
072                                  (throw (Exception. "用户名不能为空")))
073                                (when (str/blank? (:nick_name user))
074                                  (throw (Exception. "用户昵称不能为空")))
075                                (user-service/create-user! user-service
076                                                           (assoc user
077                                                                  :password default-password
078                                                                  :roles []
079                                                                  :posts []
080                                                                  :create_by (:user_name identity "")))
081                                {:user_name (:user_name user) :status "success"})
082                              (catch Exception e
083                                {:user_name (first row) :status "failed" :msg (.getMessage e)})))
084                          data-rows)
085            success-count (count (filter #(= "success" (:status %)) results))
086            failed-count (count (filter #(= "failed" (:status %)) results))]
087        (ok {:total (count results)
088             :success success-count
089             :failed failed-count
090             :details results}))
091      (catch Exception e
092        (fail (.getMessage e)))))

106  (defn export-users
107    "导出用户为 CSV 文件（带数据权限过滤）。"
108    [{:keys [user-service]} request]
109    (try
110      (let [identity (:identity request)
111            raw (:query-params request)
112            data-perm-filter (data-perm/data-perm-filter identity "default" :alias "u")
113            params (merge {:page-num 1 :page-size 10000}
114                          (dissoc raw "page" "size")
115                          (:params data-perm-filter))
116            result (user-service/list-users user-service params)
117            rows (:rows result)
118            header ["user_name" "nick_name" "email" "phonenumber" "sex" "status" "dept_id" "remark"]
119            csv-lines (mapv user->csv-row rows)
120            output (java.io.StringWriter.)]
121        (csv/write-csv output (cons header csv-lines)
122                       :separator \, :quote \")
123        (let [csv-str (str output)
124              bom "\uFEFF"
125              content (str bom csv-str)]
126          (-> (response/response content)
127              (response/header "Content-Type" "text/csv; charset=utf-8")
128              (response/header "Content-Disposition" "attachment; filename=users.csv"))))
129      (catch Exception e
130        (fail (.getMessage e)))))

148  (defn- generic-export
149    "通用导出函数。"
150    [list-fn service params header csv-fn filename request]
151    (try
152      (let [result (list-fn service (merge {:page-num 1 :page-size 10000} params))
153            rows (if (sequential? result) result (:rows result []))
154            csv-lines (mapv csv-fn rows)
155            output (java.io.StringWriter.)]
156        (csv/write-csv output (cons header csv-lines) :separator \, :quote \")
157        (let [csv-str (str output)
158              bom "\uFEFF"
159              content (str bom csv-str)]
160          (-> (response/response content)
161              (response/header "Content-Type" "text/csv; charset=utf-8")
162              (response/header "Content-Disposition" (str "attachment; filename=" filename)))))
163      (catch Exception e
164        (fail (.getMessage e)))))
