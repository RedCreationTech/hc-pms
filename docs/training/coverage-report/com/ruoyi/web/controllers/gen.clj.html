045    (when (seq content)

051  (defn- upsert-generated-sql! [table-name hugsql]
052    "将 HugSQL 写入 generated.sql，按表名替换旧块，避免重复定义。"
053    (let [file (io/file "resources/sql/generated.sql")
054          marker (str "-- == generated " table-name " ==")
055          existing (if (.exists file) (slurp file) "")
056          lines (str/split-lines existing)
057          ;; 找到本表块起止行（简单行匹配）
058          start (first (keep-indexed #(when (str/starts-with? %2 marker) %1) lines))
059          end (when start
060                (first (keep-indexed #(when (and (> %1 start)
061                                                 (str/starts-with? %2 "-- == generated ")) %1)
062                                     lines)))
063          before (if start (str/join "\n" (take start lines)) existing)
064          after (if end (str/join "\n" (drop end lines)) "")
065          block (str marker "\n" hugsql)]
066      (io/make-parents file)
067      (spit file (str (str/trim before) "\n\n" block "\n\n" (str/trim after) "\n"))
068      (.getPath file)))

073  (defn deploy-code
074    "部署生成的代码到项目目录。"
075    [{:keys [gen-service]} request]
076    (let [table-name (get-in request [:body-params :tableName])
077          code (gen-service/generate-code gen-service table-name)
078          kebab (:kebab-name code)]
079      (if (seq kebab)
080        (let [written (remove nil?
081                              [(upsert-generated-sql! table-name (:backend-sql-queries code))
082                               (write-file! (str "resources/sql/" kebab ".sql") (:backend-sql-queries code))
083                               (write-file! (str "src/clj/com/ruoyi/domain/system/" kebab ".clj") (:backend-domain code))
084                               (write-file! (str "src/clj/com/ruoyi/web/controllers/system/" kebab ".clj") (:backend-controller code))
085                               (write-file! (str "src/cljs/com/ruoyi/frontend/api/" kebab ".cljs") (:frontend-api code))
086                               (write-file! (str "src/cljs/com/ruoyi/frontend/pages/" kebab ".cljs") (:frontend-page code))
087                               (write-file! (str "src/cljs/com/ruoyi/frontend/events/" kebab ".cljs") (:frontend-events code))
088                               (write-file! (str "src/cljs/com/ruoyi/frontend/subs/" kebab ".cljs") (:frontend-subs code))
089                               (write-file! (str "resources/migrations-sqlite/" (timestamp) "_" kebab ".up.sql") (:migration-up code))
090                               (write-file! (str "resources/migrations-sqlite/" (timestamp) "_" kebab ".down.sql")
091                                            (str "DROP TABLE IF EXISTS " table-name ";"))])]
092          (ok {:message (str "代码部署成功: " table-name)
093               :written written
094               :routes (:backend-routes code)}))
095        (ok 500 "生成失败" {:error "无法生成代码"}))))

097  (defn- zip-directory! [src-dir ^File zip-file]
098    "将目录打包成 zip 文件。"
099    (with-open [zos (ZipOutputStream. (FileOutputStream. zip-file))]
100      (doseq [file (file-seq (io/file src-dir))
101              :when (.isFile file)]
102        (let [entry-name (.replace (.getPath file) (str src-dir File/separator) "")]
103          (.putNextEntry zos (ZipEntry. entry-name))
104          (io/copy file zos)
105          (.closeEntry zos))))
106    zip-file)

108  (defn- write-generated-to-dir! [base-dir code]
109    "将单个表的生成代码写入临时目录。"
110    (let [kebab (:kebab-name code)
111          ts (timestamp)
112          table-name (:table-name code)]
113      (doseq [[path content]
114              [[(str "backend/src/clj/com/ruoyi/domain/system/" kebab ".clj") (:backend-domain code)]
115               [(str "backend/src/clj/com/ruoyi/web/controllers/system/" kebab ".clj") (:backend-controller code)]
116               [(str "backend/resources/sql/" kebab ".sql") (:backend-sql-queries code)]
117               [(str "backend/resources/migrations-sqlite/" ts "_" kebab ".up.sql") (:migration-up code)]
118               [(str "backend/resources/migrations-sqlite/" ts "_" kebab ".down.sql")
119                (str "DROP TABLE IF EXISTS " table-name ";")]
120               [(str "src/cljs/com/ruoyi/frontend/api/" kebab ".cljs") (:frontend-api code)]
121               [(str "src/cljs/com/ruoyi/frontend/pages/" kebab ".cljs") (:frontend-page code)]
122               [(str "src/cljs/com/ruoyi/frontend/events/" kebab ".cljs") (:frontend-events code)]
123               [(str "src/cljs/com/ruoyi/frontend/subs/" kebab ".cljs") (:frontend-subs code)]]]
124        (when (seq content)
125          (write-file! (str base-dir File/separator path) content)))))

127  (defn download-code
128    "批量生成代码并打包成 ZIP 下载。"
129    [{:keys [gen-service]} request]
130    (let [table-names (get-in request [:body-params :tables] [])
131          codes (mapv #(gen-service/generate-code gen-service %) table-names)
132          temp-dir (io/file (str "target/gen-download-" (timestamp)))
133          zip-file (io/file (str (.getPath temp-dir) ".zip"))]
134      (if (empty? codes)
135        (ok 400 "请选择要生成的表" {})
136        (do
137          (.mkdirs temp-dir)
138          (doseq [code codes]
139            (write-generated-to-dir! (.getPath temp-dir) code))
140          (zip-directory! (.getPath temp-dir) zip-file)
141          (.deleteOnExit zip-file)
142          (-> (response/file-response (.getPath zip-file))
143              (response/content-type "application/zip")
144              (response/header "Content-Disposition" "attachment; filename=\"gen-code.zip\""))))))
