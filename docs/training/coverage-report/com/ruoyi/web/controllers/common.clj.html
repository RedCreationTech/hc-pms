001  (ns com.ruoyi.web.controllers.common
002    "通用控制器 — 文件上传、下载、资源访问。"
003    (:require
004     [clojure.java.io :as io]
005     [ring.util.response :as response]))
006  
007  (def upload-dir "uploads/")
008  (def resource-dir "resources/")
009  
010  (defn- ok
011    ([data] (ok 200 "操作成功" data))
012    ([code msg data]
013     (-> (response/response {:code code :msg msg :data data})
014         (response/content-type "application/json"))))
015  
016  (defn upload
017    "通用文件上传。"
018    [_ request]
019    (try
020      (let [dir (io/file upload-dir)]
021        (when-not (.exists dir) (.mkdirs dir))
022        (let [file (get-in request [:params :file])
023              temp-file (:tempfile file)
024              filename (:filename file)]
025          (if (and temp-file filename)
026            (let [target (io/file upload-dir filename)]
027              (io/copy temp-file target)
028              (ok {:fileName filename :url (str "/uploads/" filename)}))
029            (ok 500 "上传失败" nil))))
030      (catch Exception e (ok 500 (.getMessage e) nil))))
031  
032  (defn download
033    "通用文件下载。"
034    [_ request]
035    (let [filename (get-in request [:query-params :fileName])
036          file (io/file upload-dir filename)]
037      (if (.exists file)
038        (-> (response/response file)
039            (response/header "Content-Disposition" (str "attachment; filename=\"" filename "\""))
040            (response/content-type "application/octet-stream"))
041        (ok 404 "文件不存在" nil))))
042  
043  (defn download-resource
044    "下载资源文件。"
045    [_ request]
046    (let [resource-path (get-in request [:query-params :resource])
047          file (io/file resource-dir resource-path)]
048      (if (and resource-path (.exists file))
049        (-> (response/response file)
050            (response/header "Content-Disposition" (str "attachment; filename=\"" (.getName file) "\""))
051            (response/content-type "application/octet-stream"))
052        (ok 404 "资源不存在" nil))))
