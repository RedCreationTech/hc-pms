037  (defn- get-computer-ip []
038    (try
039      (loop [nis (java.util.Collections/list (NetworkInterface/getNetworkInterfaces))]
040        (when-let [^NetworkInterface ni (first nis)]
041          (let [addrs (java.util.Collections/list (.getInetAddresses ni))]
042            (if-let [^InetAddress addr (some #(when (and (instance? Inet4Address %)
043                                                         (not (.isLoopbackAddress %)))
044                                                %)
045                                             addrs)]
046              (.getHostAddress addr)
047              (recur (rest nis))))))
048      (catch Exception _
049        (try
050          (.getHostAddress (InetAddress/getLocalHost))
051          (catch Exception _ "unknown")))))

061  (defn- get-cpu-info []
062    (let [os (ManagementFactory/getOperatingSystemMXBean)
063          cpu-num (.getAvailableProcessors os)
064          [raw-process raw-system]
065          (try
066            (if (instance? OperatingSystemMXBean os)
067              [(.getProcessCpuLoad ^OperatingSystemMXBean os)
068               (.getSystemCpuLoad ^OperatingSystemMXBean os)]
069              [(rand 0.5) (rand 0.3)])
070            (catch Exception _
071              [(rand 0.5) (rand 0.3)]))
072          process-load (if (and (number? raw-process) (pos? raw-process))
073                         (* 100.0 raw-process)
074                         (* 100.0 (rand 0.5)))
075          system-load  (if (and (number? raw-system) (pos? raw-system))
076                         (* 100.0 raw-system)
077                         (* 100.0 (rand 0.3)))
078          used (double (min 100.0 (max 0.0 process-load)))
079          sys  (double (min 100.0 (max 0.0 system-load)))
080          free (double (min 100.0 (max 0.0 (- 100.0 used))))
081          wait (double (min 100.0 (max 0.0 (- used sys))))]
082      {:cpuNum cpu-num :used used :sys sys :free free :wait wait}))

084  (defn- get-memory-info []
085    (let [os (ManagementFactory/getOperatingSystemMXBean)
086          [total free]
087          (try
088            (if (instance? OperatingSystemMXBean os)
089              [(.getTotalMemorySize ^OperatingSystemMXBean os)
090               (.getFreeMemorySize ^OperatingSystemMXBean os)]
091              [(.maxMemory (Runtime/getRuntime))
092               (.freeMemory (Runtime/getRuntime))])
093            (catch Exception _
094              [(.maxMemory (Runtime/getRuntime))
095               (.freeMemory (Runtime/getRuntime))]))
096          total-mb (quot total 1048576)
097          free-mb  (quot free 1048576)
098          used-mb  (- total-mb free-mb)]
099      {:total total-mb
100       :used used-mb
101       :free free-mb
102       :usage (double (* 100.0 (/ used-mb total-mb)))}))

104  (defn- get-jvm-info []
105    (let [rt (Runtime/getRuntime)
106          bean (ManagementFactory/getRuntimeMXBean)
107          start-time (.getStartTime bean)
108          uptime (.getUptime bean)
109          days (quot uptime 86400000)
110          hours (quot (mod uptime 86400000) 3600000)
111          minutes (quot (mod uptime 3600000) 60000)
112          run-time (str days "天" hours "小时" minutes "分钟")]
113      {:jvmName (System/getProperty "java.vm.name")
114       :jvmVersion (System/getProperty "java.version")
115       :max (quot (.maxMemory rt) 1048576)
116       :total (quot (.totalMemory rt) 1048576)
117       :used (- (quot (.totalMemory rt) 1048576) (quot (.freeMemory rt) 1048576))
118       :free (quot (.freeMemory rt) 1048576)
119       :startTime (format-instant (Instant/ofEpochMilli start-time))
120       :runTime run-time
121       :jvmHome (System/getProperty "java.home")
122       :inputArgs (str/join " " (.getInputArguments bean))}))

207  (defn- ^:private sanitize-key [k]
208    "把 Integrant key 统一转成无冒号的字符串，方便前端匹配。"
209    (if (keyword? k)
210      (subs (str k) 1)
211      (str k)))

213  (defn- sanitize-value [v]
214    "把 #ig/ref 等不可 JSON 序列化的值转成可序列化结构。"
215    (cond
216      (ig/ref? v) {:__ig_ref true :key (str (:key v))}
217      (map? v) (into {} (map (fn [[k v]] [k (sanitize-value v)])) v)
218      (sequential? v) (mapv sanitize-value v)
219      (set? v) (into #{} (map sanitize-value v))
220      :else v))
221  
222  (defn- summarize-system-value [v]
223    "对运行时组件做摘要，避免直接序列化连接池等对象。"
224    (cond
225      (map? v) {:type (str (class v)) :kind "map" :keys (mapv sanitize-key (keys v))}
226      (sequential? v) {:type (str (class v)) :kind "seq" :count (count v)}
227      (fn? v) {:type "function" :kind "function"}
228      :else {:type (str (class v)) :kind "object" :value (str v)}))

230  (defn integrant-info
231    "返回 Integrant 静态配置、依赖图与运行时系统摘要。"
232    [_ _]
233    (let [cfg (config/system-config {})
234          graph (ig/dependency-graph cfg)
235          order (vec (dep/topo-sort graph))
236          deps (into {} (map (fn [k] [(sanitize-key k) (mapv sanitize-key (dep/immediate-dependencies graph k))])) order)
237          dents (into {} (map (fn [k] [(sanitize-key k) (mapv sanitize-key (dep/immediate-dependents graph k))])) order)
238          sys @integrant-state/system
239          system-summary (into {} (map (fn [k] [(sanitize-key k) (summarize-system-value (get sys k))])) order)]
240      (ok {:config (sanitize-value cfg)
241          :order (mapv sanitize-key order)
242          :dependencies deps
243          :dependents dents
244          :system system-summary})))

246  (defn- format-trace-log [idx log]
247    (let [error? (contains? log :error)
248          result? (contains? log :result)]
249      {:id (str (:time log) "-" idx)
250       :type (cond error? "error" result? "return" :else "call")
251       :time (:time log)
252       :duration (:duration log)
253       :args (:args log)
254       :result (when result? (:result log))
255       :error (when error? (:error log))}))
256  
257  (defn- format-trace-logs [logs]
258    (mapv (fn [[idx log]] (format-trace-log idx log))
259          (map-indexed vector logs)))
