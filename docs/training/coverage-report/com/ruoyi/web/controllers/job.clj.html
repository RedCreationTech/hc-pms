001  (ns com.ruoyi.web.controllers.job
002    "定时任务控制器。"
003    (:require
004     [clojure.string :as str]
005     [clojure.walk :as walk]
006     [com.ruoyi.infra.db :as db]
007     [com.ruoyi.infra.scheduler :as scheduler-core]
008     [com.ruoyi.infra.cron :as cron]
009     [ring.util.response :as response]))
010  
011  (defn- ok
012    ([data] (ok 200 "操作成功" data))
013    ([code msg data]
014     (-> (response/response {:code code :msg msg :data data})
015         (response/content-type "application/json"))))
016  
017  (defn- fail [msg]
018    (-> (response/response {:code 500 :msg msg})
019        (response/content-type "application/json")))
020  
021  (defn- current-user-name [request]
022    (get-in request [:identity :user-name] ""))
023  
024  (defn- body-params [request]
025    (walk/keywordize-keys (:body-params request {})))
026  
027  (defn list-jobs
028    [{:keys [query-fn]} request]
029    (ok (query-fn :list-jobs (merge {:job_name nil :job_group nil :status nil}
030                                    (:query-params request)))))
031  
032  (defn get-job
033    [{:keys [query-fn]} request]
034    (let [job-id (parse-long (get-in request [:path-params :id]))]
035      (if-let [job (query-fn :find-job-by-id {:job_id job-id})]
036        (ok job)
037        (fail "任务不存在"))))
038  
039  (defn- validate-job! [job]
040    (when-not (cron/valid? (:cron_expression job))
041      (throw (ex-info "cron 表达式不合法" {})))
042    (when-not (scheduler-core/invoke-target-allowed? (:invoke_target job))
043      (throw (ex-info "调用目标不合法或不在允许命名空间内" {}))))
044  
045  (defn create-job
046    [{:keys [query-fn db]} request]
047    (try
048      (let [params (-> {:job_name nil :job_group nil :invoke_target nil :cron_expression nil
049                        :misfire_policy nil :concurrent nil :status nil :remark nil :create_by nil}
050                       (merge (body-params request))
051                       (assoc :create_by (current-user-name request)))
052            _ (validate-job! params)
053            id (db/insert-and-get-id! query-fn db :create-job! params :last-insert-job-id :job_id)]
054        (when-let [job (query-fn :find-job-by-id {:job_id id})]
055          (scheduler-core/schedule-job! job))
056        (ok {:job_id id}))
057      (catch Exception e
058        (fail (.getMessage e)))))
059  
060  (defn update-job
061    [{:keys [query-fn]} request]
062    (try
063      (let [job-id (parse-long (get-in request [:path-params :id]))
064            params (-> {:job_name nil :job_group nil :invoke_target nil :cron_expression nil
065                        :misfire_policy nil :concurrent nil :status nil :remark nil :update_by nil}
066                       (merge (body-params request))
067                       (assoc :job_id job-id)
068                       (assoc :update_by (current-user-name request)))
069            _ (validate-job! params)]
070        (query-fn :update-job! params)
071        (when-let [job (query-fn :find-job-by-id {:job_id job-id})]
072          (scheduler-core/reschedule-job! job))
073        (ok "更新成功"))
074      (catch Exception e
075        (fail (.getMessage e)))))
076  
077  (defn delete-job
078    [{:keys [query-fn]} request]
079    (try
080      (let [job-id (parse-long (get-in request [:path-params :id]))
081            job (query-fn :find-job-by-id {:job_id job-id})]
082        (query-fn :delete-job! {:job_id job-id})
083        (when job
084          (scheduler-core/unschedule-job! job-id (:job_group job)))
085        (ok "删除成功"))
086      (catch Exception e
087        (fail (.getMessage e)))))
088  
089  (defn list-job-logs
090    [{:keys [query-fn]} request]
091    (let [params (:query-params request)
092          ->kw (fn [k] (keyword (str/replace (name k) #"-" "_")))
093          norm (->> params
094                    (map (fn [[k v]] [(->kw k) v]))
095                    (into {}))
096          page-num (or (some-> (:page_num norm) parse-long) 1)
097          page-size (or (some-> (:page_size norm) parse-long) 10)
098          offset (* (dec page-num) page-size)
099          filters (merge {:job_name nil :job_group nil :status nil}
100                         (-> norm
101                             (dissoc :page_num :page_size)
102                             (assoc :offset offset :page_size page-size)))]
103      (ok {:rows (query-fn :list-job-logs filters)
104           :total (-> (query-fn :count-job-logs filters) first :total)})))
105  
106  (defn execute-job
107    [{:keys [query-fn]} request]
108    (try
109      (let [job-id (parse-long (get-in request [:path-params :id]))]
110        (query-fn :execute-job! {:job_id job-id})
111        (ok "执行成功"))
112      (catch Exception e
113        (fail (.getMessage e)))))
114  
115  (defn change-status
116    "修改任务状态。"
117    [{:keys [query-fn]} request]
118    (let [job-id (parse-long (get-in request [:path-params :id]))
119          status (:status (body-params request))
120          job (query-fn :find-job-by-id {:job_id job-id})]
121      (query-fn :update-job! (merge {:job_id nil :job_name nil :job_group nil :invoke_target nil
122                                     :cron_expression nil :misfire_policy nil :concurrent nil
123                                     :status nil :remark nil :update_by nil}
124                                    {:job_id job-id :status status :update_by (current-user-name request)}))
125      (when job
126        (if (= "0" status)
127          (scheduler-core/resume-job! job-id (:job_group job))
128          (scheduler-core/pause-job! job-id (:job_group job))))
129      (ok "状态修改成功")))

131  (defn run-once
132    "立即执行一次任务。"
133    [_ request]
134    (let [job-id (parse-long (get-in request [:path-params :id]))]
135      (scheduler-core/trigger-job! job-id "DEFAULT")
136      (ok (str "任务 " job-id " 已触发执行"))))

139    "清空任务日志。"
140    [{:keys [query-fn]} _]
141    (query-fn :clean-job-logs! {})
142    (ok "日志已清空"))
