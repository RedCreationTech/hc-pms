001  (ns com.ruoyi.web.controllers.register
002    "用户注册控制器。"
003    (:require
004     [com.ruoyi.domain.system.user :as user-service]
005     [ring.util.response :as response]))
006  
007  (defn- ok
008    ([msg] (-> (response/response {:code 200 :msg msg})
009               (response/content-type "application/json")))
010    ([code msg] (-> (response/response {:code code :msg msg})
011                    (response/content-type "application/json"))))
012  
013  (defn register
014    "用户注册。"
015    [{:keys [user-service]} request]
016    (try
017      (let [params (:body-params request)
018            username (:username params)]
019        (if-let [existing (user-service/find-user-by-name user-service username)]
020          (ok 500 "注册账号已存在")
021          (do (user-service/create-user! user-service (assoc params :user_name username))
022              (ok "注册成功"))))
023      (catch Exception e (ok 500 (.getMessage e)))))
