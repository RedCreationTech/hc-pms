045  ;; log uncaught exceptions in threads
046  (Thread/setDefaultUncaughtExceptionHandler
047   (fn [thread ex]
048     (log/error {:what :uncaught-exception
049                 :exception ex
050                 :where (str "Uncaught exception on" (.getName thread))})))

052  (def system integrant-state/system)
053  
054  (defn stop-app []
055    ((or (:stop defaults) (fn [])))
056    (some-> (deref system) (ig/halt!)))
057  
058  (defn start-app [& [params]]
059    ((or (:start params) (:start defaults) (fn [])))
060    (->> (config/system-config (or (:opts params) (:opts defaults) {}))
061         (ig/expand)
062         (ig/init)
063         (reset! system)))
064  
065  (defn -main [& _]
066    (start-app)
067    (.addShutdownHook (Runtime/getRuntime) (Thread. (fn [] (stop-app) (shutdown-agents))))
068    ;; Keep main thread alive so JVM doesn't exit
069    (while true (Thread/sleep 60000)))
