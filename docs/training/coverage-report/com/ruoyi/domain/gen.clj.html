009  (defn list-tables
010    "查询数据库中的所有表。"
011    [{:keys [query-fn db]}]
012    (try
013      (if db
014        (idb/get-tables db)
015        (query-fn :gen-tables {}))
016      (catch Exception _ [])))

018  (defn table-columns
019    "查询指定表的列信息。"
020    [{:keys [query-fn db]} table-name]
021    (try
022      (if db
023        (idb/get-table-columns db table-name)
024        (query-fn :gen-columns {:table-name table-name}))
025      (catch Exception _ [])))

029  (defn- col-name [col]
030    (or (:name col) (:column_name col)))
031  
032  (defn- col-type [col]
033    (or (:type col) (:data_type col)))
034  
035  (defn- col-pk? [col]
036    (or (= 1 (:pk col))
037        (= "YES" (:is_pk col))))
038  
039  (defn- col-nullable? [col]
040    (or (= 0 (:notnull col))
041        (= "YES" (:is_nullable col))))

060  (defn- sql-type [db-type]
061    (let [t (str/lower-case (str db-type))]
062      (cond
063        (str/starts-with? t "int") "INTEGER"
064        (or (str/starts-with? t "varchar")
065            (str/starts-with? t "char")
066            (str/starts-with? t "text")
067            (str/starts-with? t "datetime")
068            (str/starts-with? t "date")) "TEXT"
069        (str/starts-with? t "real") "REAL"
070        (str/starts-with? t "blob") "BLOB"
071        :else "TEXT")))

073  (defn- generate-ddl [table-name columns]
074    (let [lines (mapv (fn [col]
075                        (str "    " (col-name col) " " (sql-type (col-type col))
076                             (when (col-pk? col) " PRIMARY KEY AUTOINCREMENT")
077                             (when (not (col-nullable? col)) " NOT NULL")
078                             (when-let [d (:dflt_value col)]
079                               (str " DEFAULT " d))))
080                      columns)]
081      (str "CREATE TABLE IF NOT EXISTS " table-name " (\n"
082           (str/join ",\n" lines)
083           "\n);")))
084  
085  (defn- generate-hugsql [table-name kebab columns]
086    (let [pk-col (or (first (filter col-pk? columns)) (first columns))
087          pk (col-name pk-col)
088          non-pk (remove #(= (col-name %) pk) columns)
089          list-fn (str "list-" kebab "s")
090          find-fn (str "find-" kebab "-by-id")
091          create-fn (str "create-" kebab "!")
092          update-fn (str "update-" kebab "!")
093          delete-fn (str "delete-" kebab "!")
094          col-list (str/join ", " (map col-name columns))
095          insert-cols (str/join ", " (map col-name non-pk))
096          insert-params (str/join ", " (map #(str ":" (col-name %)) non-pk))
097          set-clause (str/join ",\n" (map #(str "    " (col-name %) " = COALESCE(:" (col-name %) ", " (col-name %) ")") non-pk))]
098      (str "-- :name " list-fn "\n"
099           "-- :command :query\n"
100           "-- :result :*\n"
101           "SELECT " col-list "\n  FROM " table-name "\n ORDER BY " pk " DESC;\n\n"
102           "-- :name " find-fn "\n"
103           "-- :command :query\n"
104           "-- :result :one\n"
105           "SELECT " col-list "\n  FROM " table-name "\n WHERE " pk " = :" pk ";\n\n"
106           "-- :name " create-fn "\n"
107           "-- :command :insert\n"
108           "INSERT INTO " table-name " (" insert-cols ")\n"
109           "VALUES (" insert-params ");\n\n"
110           "-- :name " update-fn "\n"
111           "-- :command :execute\n"
112           "UPDATE " table-name "\n"
113           "   SET " set-clause "\n"
114           " WHERE " pk " = :" pk ";\n\n"
115           "-- :name " delete-fn "\n"
116           "-- :command :execute\n"
117           "DELETE FROM " table-name "\n"
118           " WHERE " pk " = :" pk ";\n")))

246  (defn generate-code
247    "根据表名生成完整 CRUD 代码模板。"
248    [{:keys [query-fn] :as ctx} table-name]
249    (let [columns (table-columns ctx table-name)
250          entity (entity-name table-name)
251          kebab (kebab-name table-name)
252          pk-name (col-name (or (first (filter col-pk? columns)) (first columns)))]
253      (merge
254       {:table-name table-name
255        :entity-name entity
256        :kebab-name kebab
257        :columns (count columns)
258        :message (str "已生成 " table-name " 的 CRUD 代码")}
259       (when (seq columns)
260         {:backend-sql (generate-ddl table-name columns)
261          :backend-sql-queries (generate-hugsql table-name kebab columns)
262          :backend-domain (generate-domain table-name entity kebab columns)
263          :backend-controller (generate-controller entity kebab pk-name)
264          :backend-routes (generate-routes kebab)
265          :frontend-api (generate-frontend-api kebab)
266          :frontend-page (generate-frontend-page kebab entity)
267          :frontend-events (generate-frontend-events kebab)
268          :frontend-subs (generate-frontend-subs kebab)
269          :migration-up (generate-ddl table-name columns)}))))
