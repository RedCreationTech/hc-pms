001  (ns com.ruoyi.domain.system.form-template
002    "表单模板领域服务。"
003    (:require
004     [com.ruoyi.infra.db :as db]))
005  
006  (defn list-form-templates
007    "查询表单模板列表。"
008    [{:keys [query-fn]} params]
009    (query-fn :list-form-templates (merge {:form_name nil :form_key nil} params)))
010  
011  (defn find-form-template-by-id
012    "根据ID查询表单模板。"
013    [{:keys [query-fn]} id]
014    (query-fn :find-form-template-by-id {:id id}))
015  
016  (defn find-form-template-by-key
017    "根据form_key查询表单模板。"
018    [{:keys [query-fn]} form-key]
019    (query-fn :find-form-template-by-key {:form_key form-key}))
020  
021  (defn create-form-template!
022    "创建表单模板，返回自增ID。"
023    [{:keys [query-fn db]} params]
024    (when (find-form-template-by-key {:query-fn query-fn} (:form_key params))
025      (throw (ex-info "表单key已存在" {:form_key (:form_key params)})))
026    (db/insert-and-get-id! query-fn db :create-form-template!
027                           (merge {:form_name nil :form_key nil :schema_json nil :remark nil :create_by nil} params)))
028  
029  (defn update-form-template!
030    "更新表单模板。"
031    [{:keys [query-fn]} params]
032    (let [id (:id params)
033          new-key (:form_key params)
034          existing (when new-key (find-form-template-by-key {:query-fn query-fn} new-key))]
035      (when (and existing (not= (:id existing) id))
036        (throw (ex-info "表单key已存在" {:form_key new-key})))
037      (query-fn :update-form-template!
038                (merge {:id id :form_name nil :form_key nil :schema_json nil :remark nil :update_by nil} params))))
039  
040  (defn delete-form-template!
041    "删除表单模板。"
042    [{:keys [query-fn]} id]
043    (query-fn :delete-form-template! {:id id}))
