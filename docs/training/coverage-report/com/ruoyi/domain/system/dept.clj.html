001  (ns com.ruoyi.domain.system.dept
002    "部门领域服务。"
003    (:require
004     [clojure.string :as str]
005     [com.ruoyi.infra.db :as db]))
006  
007  (defn list-depts
008    "查询部门列表。"
009    [{:keys [query-fn]} params]
010    (query-fn :list-depts (merge {:status nil :dept_name nil} params)))
011  
012  (defn find-dept-by-id
013    "根据ID查询部门。"
014    [{:keys [query-fn]} dept-id]
015    (query-fn :find-dept-by-id {:dept_id dept-id}))
016  
017  (defn- compute-ancestors
018    "根据父部门计算 ancestors 路径。"
019    [{:keys [query-fn]} parent-id]
020    (if (and parent-id (pos? parent-id))
021      (if-let [parent (query-fn :find-dept-by-id {:dept_id parent-id})]
022        (let [parent-anc (or (:ancestors parent) "0")]
023          (if (seq parent-anc)
024            (str parent-anc "," parent-id)
025            (str "0," parent-id)))
026        (str "0," parent-id))
027      "0"))
028  
029  (defn- update-descendants-ancestors!
030    "递归更新子部门的 ancestors。"
031    [{:keys [query-fn]} dept-id ancestors]
032    (let [children (query-fn :list-depts-by-parent {:parent_id dept-id})]
033      (doseq [child children]
034        (let [child-id (:dept_id child)
035              child-ancestors (str ancestors "," child-id)]
036          (query-fn :update-dept-ancestors! {:dept_id child-id :ancestors child-ancestors})
037          (update-descendants-ancestors! {:query-fn query-fn} child-id child-ancestors)))))
038  
039  (defn create-dept!
040    "创建部门。"
041    [{:keys [query-fn db] :as ctx} params]
042    (let [parent-id (:parent_id params 0)
043          params (-> {:parent_id nil :ancestors nil :dept_name nil :order_num nil
044                      :leader nil :phone nil :email nil :status nil :create_by nil}
045                     (merge params)
046                     (assoc :parent_id parent-id
047                            :ancestors (compute-ancestors ctx parent-id)))]
048      (db/insert-and-get-id! query-fn db :create-dept! params)))
049  
050  (defn update-dept!
051    "更新部门。"
052    [{:keys [query-fn] :as ctx} params]
053    (let [dept-id (:dept_id params)
054          parent-id (:parent_id params)
055          params (merge {:parent_id nil :ancestors nil :dept_name nil :order_num nil
056                         :leader nil :phone nil :email nil :status nil :update_by nil}
057                        params)]
058      (if (some? parent-id)
059        (let [ancestors (compute-ancestors ctx parent-id)]
060          (query-fn :update-dept! (assoc params :ancestors ancestors))
061          (update-descendants-ancestors! ctx dept-id ancestors))
062        (query-fn :update-dept! params))))
063  
064  (defn delete-dept!
065    "逻辑删除部门。"
066    [{:keys [query-fn]} dept-id]
067    (query-fn :delete-dept! {:dept_id dept-id}))
