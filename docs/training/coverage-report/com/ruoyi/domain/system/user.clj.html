001  (ns com.ruoyi.domain.system.user
002    "用户领域服务，处理用户 CRUD、密码管理与角色关联。"
003    (:require
004     [com.ruoyi.infra.security :as security]
005     [clojure.string :as str]
006     [clojure.tools.logging :as log]
007     [com.ruoyi.infra.db :as db]))
008  
009  (defn list-users
010    "查询用户列表，支持分页和条件筛选。"
011    [{:keys [query-fn]} params]
012    (let [page-num (or (:page-num params) 1)
013          page-size (or (:page-size params) 10)
014          offset (* (dec page-num) page-size)
015          filters (merge {:user_name nil :phonenumber nil :status nil
016                          :dept_id nil :params nil}
017                         (-> params
018                             (dissoc :page-num :page-size))
019                         {:offset offset :page_size page-size})
020          rows (query-fn :list-users filters)
021          total (query-fn :count-users filters)]
022      {:rows rows :total (:total total)}))
023  
024  (defn find-user-by-id
025    "根据ID查询用户详情，包含部门、角色、岗位信息。"
026    [{:keys [query-fn]} user-id]
027    (when-let [user (query-fn :find-user-by-id {:user_id user-id})]
028      (assoc user
029             :roles (query-fn :list-roles-by-user-id {:user_id user-id})
030             :posts (query-fn :list-posts-by-user-id {:user_id user-id}))))
031  
032  (defn find-user-by-name
033    "根据用户名查询用户（用于登录）。"
034    [{:keys [query-fn]} user-name]
035    (query-fn :find-user-by-name {:user_name user-name}))
036  
037  (defn create-user!
038    "创建新用户，自动加密密码。"
039    [{:keys [query-fn db]} {:keys [password roles posts] :as params}]
040    (let [hashed (security/hash-password password)]
041      (let [user-id (db/insert-and-get-id! query-fn db :create-user!
042                                           (-> params
043                                               (assoc :password hashed)
044                                               (dissoc :roles :posts)))]
045        ;; 关联角色
046        (doseq [role-id roles]
047          (query-fn :insert-user-role! {:user_id user-id :role_id role-id}))
048        ;; 关联岗位
049        (doseq [post-id posts]
050          (query-fn :insert-user-post! {:user_id user-id :post_id post-id}))
051        user-id)))
052  
053  (defn update-user!
054    "更新用户信息，可选更新密码。"
055    [{:keys [query-fn]} {:keys [user-id password roles posts] :as params}]
056    (let [update-data (-> params
057                          (dissoc :roles :posts :user-id)
058                          (assoc :user_id user-id))
059          update-data (if password
060                        (assoc update-data :password (security/hash-password password))
061                        update-data)]
062      (query-fn :update-user! update-data)
063      ;; 更新角色关联
064      (when roles
065        (query-fn :delete-user-roles! {:user_id user-id})
066        (doseq [role-id roles]
067          (query-fn :insert-user-role! {:user_id user-id :role_id role-id})))
068      ;; 更新岗位关联
069      (when posts
070        (query-fn :delete-user-posts! {:user_id user-id})
071        (doseq [post-id posts]
072          (query-fn :insert-user-post! {:user_id user-id :post_id post-id})))
073      user-id))
074  
075  (defn delete-user!
076    "逻辑删除用户。"
077    [{:keys [query-fn]} user-id]
078    (query-fn :delete-user! {:user_id user-id}))
079  
080  (defn get-user-roles
081    "获取用户角色列表。"
082    [{:keys [query-fn]} user-id]
083    (query-fn :list-roles-by-user-id {:user_id user-id}))
084  
085  (defn update-user-roles!
086    "更新用户角色（先删后插）。"
087    [{:keys [query-fn]} {:keys [user-id role-ids]}]
088    (query-fn :delete-user-roles! {:user_id user-id})
089    (doseq [rid role-ids]
090      (query-fn :insert-user-role! {:user_id user-id :role_id rid})))
