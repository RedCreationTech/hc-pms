001  (ns com.ruoyi.domain.system.menu
002    "菜单领域服务，处理菜单树构建与 CRUD。"
003    (:require
004     [clojure.walk :as walk]
005     [com.ruoyi.infra.db :as db]))
006  
007  (defn list-menus
008    "查询菜单列表。"
009    [{:keys [query-fn]} params]
010    (query-fn :list-menus (merge {:menu_name nil :status nil :menu_type nil} params)))
011  
012  (defn find-menu-by-id
013    "根据ID查询菜单。"
014    [{:keys [query-fn]} menu-id]
015    (query-fn :find-menu-by-id {:menu_id menu-id}))
016  
017  (defn create-menu!
018    "创建菜单。"
019    [{:keys [query-fn db]} params]
020    (db/insert-and-get-id! query-fn db :create-menu!
021                           (merge {:menu_name nil :parent_id nil :order_num nil :path nil
022                                   :component nil :query nil :route_name nil :is_frame nil
023                                   :is_cache nil :menu_type nil :visible nil :status nil
024                                   :perms nil :icon nil :create_by nil}
025                                  params)))
026  
027  (defn update-menu!
028    "更新菜单。"
029    [{:keys [query-fn]} params]

031                                    :component nil :query nil :route_name nil :is_frame nil
032                                    :is_cache nil :menu_type nil :visible nil :status nil
033                                    :perms nil :icon nil :create_by nil :create_time nil :update_by nil}
034                                   params)))
035  
036  (defn delete-menu!
037    "删除菜单。"
038    [{:keys [query-fn]} menu-id]
039    (query-fn :delete-menu! {:menu_id menu-id}))
040  
041  (defn- build-tree
042    "将扁平菜单列表构建为树形结构。"
043    [items parent-id]
044    (let [children (filter #(= (:parent_id %) parent-id) items)]
045      (mapv (fn [child]
046              (assoc child :children (build-tree items (:menu_id child))))
047            children)))
048  
049  (defn menu-tree
050    "获取菜单树。"
051    [{:keys [query-fn]}]
052    (let [menus (query-fn :list-menus {:menu_name nil :status nil :menu_type nil})]
053      (build-tree menus 0)))
054  
055  (defn menu-tree-by-roles
056    "根据角色ID列表构建菜单树。"
057    [{:keys [query-fn]} role-ids]
058    (if (seq role-ids)
059      (let [menus (query-fn :list-menus-by-role-ids {:role-ids role-ids})]
060        (build-tree menus 0))
061      []))
