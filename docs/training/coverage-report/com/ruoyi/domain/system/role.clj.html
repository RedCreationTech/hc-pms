001  (ns com.ruoyi.domain.system.role
002    "角色领域服务，处理角色 CRUD、菜单授权与数据权限。"
003    (:require
004     [clojure.set :as set]
005     [com.ruoyi.infra.db :as db]))
006  
007  (defn list-roles
008    "查询角色列表。"
009    [{:keys [query-fn]} params]
010    (query-fn :list-roles (merge {:role_name nil :role_key nil :status nil} params)))
011  
012  (defn find-role-by-id
013    "根据ID查询角色详情，包含关联菜单ID列表。"
014    [{:keys [query-fn]} role-id]
015    (when-let [role (query-fn :find-role-by-id {:role_id role-id})]
016      (assoc role :menu-ids (mapv :menu_id (query-fn :list-menus-by-role-id {:role_id role-id})))))
017  
018  (defn create-role!
019    "创建角色并绑定菜单权限。"
020    [{:keys [query-fn db]} {:keys [menu-ids] :as params}]
021    (let [role-id (db/insert-and-get-id! query-fn db :create-role! (dissoc params :menu-ids))]
022      (doseq [m-id menu-ids]
023        (query-fn :insert-role-menu! {:role_id role-id :menu_id m-id}))
024      role-id))
025  
026  (defn update-role!
027    "更新角色及菜单权限。"
028    [{:keys [query-fn]} {:keys [role-id menu-ids] :as params}]
029    ;; 更新角色基本信息（只保留实际字段）
030    (let [role-params (merge {:role_name nil :role_key nil :role_sort nil :data_scope nil
031                              :menu_check_strictly nil :dept_check_strictly nil
032                              :status nil :remark nil}
033                             (select-keys params [:role_name :role_key :role_sort :data_scope
034                                                  :menu_check_strictly :dept_check_strictly
035                                                  :status :remark])
036                             {:role_id role-id})]
037      (query-fn :update-role! role-params))
038    ;; 更新菜单权限
039    (when menu-ids
040      (query-fn :delete-role-menus! {:role_id role-id})
041      (doseq [m-id menu-ids]
042        (query-fn :insert-role-menu! {:role_id role-id :menu_id (if (string? m-id) (parse-long m-id) m-id)})))
043    role-id)
044  
045  (defn delete-role!
046    "逻辑删除角色。"
047    [{:keys [query-fn]} role-id]
048    (query-fn :delete-role! {:role_id role-id}))
049  
050  (defn list-allocated-users
051    "查询已分配该角色的用户列表。"
052    [{:keys [query-fn]} {:keys [role-id user-name phonenumber]}]
053    (query-fn :list-users-by-role {:role_id role-id :user_name user-name :phonenumber phonenumber}))
054  
055  (defn list-unallocated-users
056    "查询未分配该角色的用户列表。"
057    [{:keys [query-fn]} {:keys [role-id user-name phonenumber]}]
058    (query-fn :list-users-not-in-role {:role_id role-id :user_name user-name :phonenumber phonenumber}))
059  
060  (defn cancel-auth-user!
061    "取消用户角色授权。"
062    [{:keys [query-fn]} {:keys [role-id user-id]}]
063    (query-fn :delete-user-role! {:role_id role-id :user_id user-id}))
064  
065  (defn cancel-auth-user-all!
066    "批量取消用户角色授权。"
067    [{:keys [query-fn]} {:keys [role-id user-ids]}]
068    (doseq [uid user-ids]
069      (query-fn :delete-user-role! {:role_id role-id :user_id uid})))
070  
071  (defn select-auth-user-all!
072    "批量授权用户角色（批量插入）。"
073    [{:keys [query-fn]} {:keys [role-id user-ids]}]
074    (doseq [uid user-ids]
075      (query-fn :insert-user-role! {:role_id role-id :user_id uid})))
076  
077  (defn dept-tree-by-role
078    "获取角色关联的部门树。"
079    [{:keys [query-fn]} dept-service role-id]
080    (let [role (query-fn :find-role-by-id {:role_id role-id})
081          depts (dept-service (:list-depts dept-service))]
082      {:depts depts :checked-keys (when role [(:dept_ids role)])}))
083  
084  (defn get-role-perms
085    "获取角色的所有权限标识。"
086    [{:keys [query-fn]} role-id]
087    (->> (query-fn :list-menus-by-role-id {:role_id role-id})
088         (map :perms)
089         (remove nil?)
090         (remove empty?)
091         (into #{})))
