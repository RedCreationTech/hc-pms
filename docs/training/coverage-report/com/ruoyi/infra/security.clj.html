001  (ns com.ruoyi.infra.security
002    "安全工具模块，提供密码哈希与 JWT 令牌签发/验证功能。"
003    (:require
004     [buddy.hashers :as hashers]
005     [buddy.sign.jwt :as jwt]
006     [clojure.string :as str]))
007  
008  (def secret-key
009    "JWT 签名密钥，生产环境应通过环境变量注入。"
010    (or (System/getenv "JWT_SECRET")
011        "rouyi-default-jwt-secret-key-change-in-production"))
012  
013  (defn hash-password
014    "使用 bcrypt 对明文密码进行哈希。"
015    [plain-text]
016    (hashers/derive plain-text {:alg :bcrypt+blake2b-512}))
017  
018  (defn verify-password
019    "验证明文密码与哈希值是否匹配。"
020    [plain-text hashed]
021    (hashers/check plain-text hashed))
022  
023  (defn generate-token
024    "为用户生成 JWT 访问令牌，包含用户ID、用户名和角色列表。"
025    [user-id user-name roles & {:keys [exp-hours]
026                                :or {exp-hours 24}}]
027    (let [claims {:user-id user-id
028                  :user-name user-name
029                  :roles roles
030                  :exp (+ (System/currentTimeMillis)
031                          (* exp-hours 60 60 1000))}]
032      (jwt/sign claims secret-key {:alg :hs256})))
033  
034  (defn parse-token
035    "解析并验证 JWT 令牌，成功返回 claims，失败返回 nil。"
036    [token]
037    (try
038      (jwt/unsign token secret-key {:alg :hs256})
039      (catch Exception _
040        nil)))
041  
042  (defn extract-token
043    "从 Authorization Header 中提取 Bearer Token。"
044    [request]
045    (some-> (get-in request [:headers "authorization"])
046            (str/replace-first #"(?i)^Bearer\s+" "")))
