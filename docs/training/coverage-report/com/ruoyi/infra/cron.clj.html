001  (ns com.ruoyi.infra.cron
002    "Cron 表达式辅助工具。"
003    (:require
004     [clojure.string :as str])
005    (:import
006     [org.quartz CronExpression]
007     [org.quartz CronScheduleBuilder]))
008  
009  (defn- normalize [expression]
010    "将 5 字段 Unix cron 补全为 Quartz 6 字段（秒位为 0）。"
011    (if (and (seq expression)
012             (= 5 (count (str/split expression #"\s+"))))
013      (str "0 " expression)
014      expression))
015  
016  (defn valid?
017    "校验 cron 表达式是否合法（Quartz 格式，5-7 位；5 位自动补秒）。"
018    [expression]
019    (and (seq expression)
020         (CronExpression/isValidExpression (normalize expression))))
021  
022  (defn cron-schedule
023    "根据 cron 表达式和 misfire 策略构建 CronScheduleBuilder。"
024    [expression misfire-policy]
025    (let [builder (CronScheduleBuilder/cronSchedule ^String (normalize expression))]
026      (case (when misfire-policy (str misfire-policy))
027        "1" (.withMisfireHandlingInstructionIgnoreMisfires builder)
028        "2" (.withMisfireHandlingInstructionFireAndProceed builder)
029        "3" (.withMisfireHandlingInstructionDoNothing builder)
030        builder)))
