043    "com.ruoyi.task")

045  (defn invoke-target-allowed?
046    "校验 invoke_target 不包含危险协议，且调用目标在允许命名空间内。"
047    [target]
048    (and (not (some #(str/includes? target %) dangerous-patterns))
049         (str/starts-with? target allowed-ns-prefix)))

051  (defn- parse-invoke-target
052    "解析 invoke_target 字符串。
053  
054    支持两种形式：
055    - task-ns/task-fn           (无参函数)
056    - task-ns/task-fn('a', 1)   (带参函数，仅支持字符串与数字字面量)
057  
058    返回 {:ns :fn :args}"
059    [target]
060    (when (seq target)
061      (let [[head args-str] (str/split target #"\(" 2)
062            [ns-part fn-part] (if (str/includes? head "/")
063                                (str/split head #"/" 2)
064                                [allowed-ns-prefix head])]
065        {:ns (symbol ns-part)
066         :fn (symbol fn-part)
067         :args (when args-str
068                 (-> args-str
069                     (str/replace #"\)$" "")
070                     (str/split #"\s*,\s*")))})))
071  
072  (defn- parse-arg
073    "将参数字符串解析为 Clojure 值。"
074    [s]
075    (let [s (str/trim s)]
076      (cond
077        (re-matches #"^['\"].*['\"]$" s) (str/replace s #"^['\"]|['\"]$" "")
078        (re-matches #"^-?\d+$" s) (parse-long s)
079        (re-matches #"^-?\d+\.\d+$" s) (parse-double s)
080        (= "true" s) true
081        (= "false" s) false
082        :else s)))
083  
084  (defn- invoke-target!
085    "执行 invoke_target 指向的函数。"
086    [target]
087    (if-let [{:keys [ns fn args]} (parse-invoke-target target)]
088      (do
089        (require ns)
090        (let [var (ns-resolve ns fn)
091              parsed-args (mapv parse-arg args)]
092          (if var
093            (apply @var parsed-args)
094            (throw (ex-info (str "Target function not found: " target) {})))))
095      (throw (ex-info (str "Invalid invoke target: " target) {}))))

097  (defn- write-job-log!
098    "记录任务执行日志。"
099    [job status message exception]
100    (try
101      ((query-fn) :create-job-log!
102                  {:job_name (:job_name job)
103                   :job_group (:job_group job)
104                   :invoke_target (:invoke_target job)
105                   :job_message message
106                   :status status
107                   :exception_info (or (some-> exception Throwable->map str) "")})
108      (catch Exception e
109        (log/warn e "Failed to write job log"))))

111  (def ^Job ruoyi-job
112    (proxy [Job] []
113      (execute [^JobExecutionContext ctx]
114        (let [data (.getMergedJobDataMap ctx)
115              job {:job_id (.getString data "job_id")
116                   :job_name (.getString data "job_name")
117                   :job_group (.getString data "job_group")
118                   :invoke_target (.getString data "invoke_target")}
119              start (System/currentTimeMillis)]
120          (try
121            (invoke-target! (:invoke_target job))
122            (write-job-log! job "0"
123                            (str "任务执行成功，耗时 " (- (System/currentTimeMillis) start) " ms")
124                            nil)
125            (catch Exception e
126              (log/warn e "Job execution failed" job)
127              (write-job-log! job "1"
128                              (str "任务执行失败，耗时 " (- (System/currentTimeMillis) start) " ms")
129                              e)))
130          nil))))

134  (defn- job-key [job-id job-group]
135    (JobKey/jobKey (str "job_" job-id) (or job-group "DEFAULT")))

137  (defn- trigger-key [job-id job-group]
138    (TriggerKey/triggerKey (str "trigger_" job-id) (or job-group "DEFAULT")))

140  (defn- build-job-detail [job]
141    (-> (JobBuilder/newJob (class ruoyi-job))
142        (.withIdentity (job-key (:job_id job) (:job_group job)))
143        (.usingJobData "job_id" (str (:job_id job)))
144        (.usingJobData "job_name" (str (:job_name job)))
145        (.usingJobData "job_group" (str (:job_group job)))
146        (.usingJobData "invoke_target" (str (:invoke_target job)))
147        (.build)))
148  
149  (defn- build-cron-trigger [job]
150    (-> (TriggerBuilder/newTrigger)
151        (.withIdentity (trigger-key (:job_id job) (:job_group job)))
152        (.forJob (job-key (:job_id job) (:job_group job)))
153        (.withSchedule (cron/cron-schedule (:cron_expression job)
154                                           (:misfire_policy job)))
155        (.build)))

157  (defn schedule-job!
158    "将 sys_job 记录注册到 Quartz 调度器。"
159    [job]
160    (when (and (seq (:cron_expression job)) (invoke-target-allowed? (:invoke_target job)))
161      (let [s (scheduler)
162            job-detail (build-job-detail job)
163            trigger (build-cron-trigger job)]
164        (.scheduleJob s job-detail trigger)
165        (when (= "1" (:status job))
166          (.pauseJob s (job-key (:job_id job) (:job_group job)))))))
167  
168  (defn reschedule-job!
169    "更新 Quartz 中的任务。"
170    [job]
171    (let [s (scheduler)
172          k (job-key (:job_id job) (:job_group job))]
173      (when (.checkExists s k)
174        (.deleteJob s k))
175      (schedule-job! job)))

201  (defn load-jobs!
202    "从 sys_job 表加载所有正常任务到调度器。"
203    []
204    (let [jobs ((query-fn) :list-jobs {:job_name nil :job_group nil :status nil})]
205      (doseq [job jobs]
206        (try
207          (schedule-job! job)
208          (catch Exception e
209            (log/warn e "Failed to schedule job" job))))))
