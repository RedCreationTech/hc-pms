001  (ns com.ruoyi.infra.data-perm
002    "数据权限过滤。
003    
004    支持 RuoYi 的五种数据权限范围:
005    - 1: ALL          — 全部数据
006    - 2: CUSTOM       — 自定义 (角色关联的部门)
007    - 3: DEPT         — 本部门
008    - 4: DEPT_CHILD   — 本部门及以下
009    - 5: SELF         — 仅本人
010  
011    用法: (data-perm-filter identity role-key table-alias)
012    返回一个 SQL 片段或 WHERE 条件 map，注入到 HugSQL 查询中。"
013    (:require
014     [clojure.string :as str]
015     [clojure.tools.logging :as log]))
016  
017  ;; ──────────── 数据权限 SQL 片段生成 ────────────
018  
019  (def scope-names
020    "数据权限范围名称映射"
021    {1 "ALL"
022     2 "CUSTOM"
023     3 "DEPT"
024     4 "DEPT_CHILD"
025     5 "SELF"})
026  
027  (defn- build-dept-filter
028    "根据角色数据权限范围，生成部门过滤条件。
029    返回 {:dept_ids [...] :user_id ...} 形式的参数 map 和 SQL where 片段。"
030    [role dept-ids alias user-id]
031    (let [scope (:data-scope role 5)
032          alias  (or alias "u")]
033      (case scope
034        ;; 1: ALL — 不追加过滤
035        1 {:params {} :sql "1=1"}
036  
037        ;; 2: CUSTOM — 自定义部门范围
038        2 (let [ids (or dept-ids [(:dept-id role)])]
039            {:params {:data-perm-dept-ids ids}
040             :sql    (str " " alias ".dept_id IN (:v*:data-perm-dept-ids)")})
041  
042        ;; 3: DEPT — 本部门
043        3 (if-let [dept-id (:dept-id role)]
044            {:params {:data-perm-dept-id dept-id}
045             :sql    (str " " alias ".dept_id = :data-perm-dept-id")}
046            {:params {} :sql "1=0"})
047  
048        ;; 4: DEPT_CHILD — 本部门及以下
049        4 (if-let [dept-id (:dept-id role)]
050            {:params {:data-perm-dept-id dept-id}
051             :sql    (str " " alias ".dept_id IN (SELECT dept_id FROM sys_dept "
052                          "WHERE del_flag = '0' AND (dept_id = :data-perm-dept-id OR ancestors LIKE '%/' || :data-perm-dept-id || '/%' || '%'))")}
053            {:params {} :sql "1=0"})
054  
055        ;; 5: SELF — 仅本人
056        5 (if-let [uid user-id]
057            {:params {:data-perm-user-id uid}
058             :sql    (str " " alias ".user_id = :data-perm-user-id")}
059            {:params {} :sql "1=0"}))))
060  
061  (defn data-perm-filter
062    "为查询生成数据权限过滤条件。
063    
064    identity  — 当前用户 claims (含 :user-id :roles)
065    role-key  — 查询所用的角色标识 (如 \"admin\" 不过滤)
066    alias     — SQL 表别名 (默认 \"u\")
067    
068    返回 {:sql \"...\" :params {...}} 合并到查询参数中。"
069    [identity role-key & {:keys [dept-ids alias]
070                          :or   {alias "u"}}]
071    (let [user-id  (:user-id identity)
072          user-name (:user-name identity)
073          roles    (:roles identity [])]
074      ;; 管理员不过滤
075      (if (some #(= "admin" (:role-key %)) roles)
076        {:params {} :sql "1=1"}
077        (let [role (first (filter #(= role-key (:role-key %)) roles))]
078          (if role
079            (build-dept-filter role dept-ids alias user-id)
080            {:params {} :sql "1=1"})))))
