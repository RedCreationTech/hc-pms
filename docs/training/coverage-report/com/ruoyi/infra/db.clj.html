009  (defn detect-db-type
010    "检测数据库类型。支持 DataSource、Connection 以及 next.jdbc 包装对象。"
011    [db]
012    (let [connable (or (:connectable db) db)]
013      (try
014        (let [product-name (try
015                             (.getDatabaseProductName (.getMetaData connable))
016                             (catch Exception _
017                               (with-open [conn (jdbc/get-connection connable)]
018                                 (.getDatabaseProductName (.getMetaData conn)))))]
019          (cond
020            (str/includes? (str/lower-case product-name) "sqlite") :sqlite
021            (str/includes? (str/lower-case product-name) "mysql") :mysql
022            :else :unknown))
023        (catch Exception _ :unknown))))
024  
025  (defn- connectable
026    "提取可用于 JDBC 执行的数据源或连接。"
027    [db]
028    (or (:connectable db) db))

030  (defn last-insert-id
031    "获取最近一次插入的自增 ID，自动适配 SQLite/MySQL。
032     默认使用 :last-insert-rowid（SQLite）或 :last-insert-rowid-mysql（MySQL）查询。
033     可通过 result-key 指定返回字段名，例如 :job_id。
034     注意：MySQL 下请传入与插入同一事务的连接，否则可能获取不到 ID。"
035    ([query-fn db]
036     (last-insert-id query-fn db :last-insert-rowid :last_insert_rowid))
037    ([query-fn db query-name result-key]
038     (let [db-type (detect-db-type db)
039           q (if (= :mysql db-type)
040               (keyword (str (name query-name) "-mysql"))
041               query-name)]
042       (get (query-fn db q {}) result-key))))
043  
044  (defn insert-and-get-id!
045    "在同一事务中执行插入并返回自增 ID，自动适配 SQLite/MySQL。"
046    ([query-fn db insert-query params]
047     (insert-and-get-id! query-fn db insert-query params :last-insert-rowid :last_insert_rowid))
048    ([query-fn db insert-query params id-query id-key]
049     (let [db-type (detect-db-type db)
050           id-q (if (= :mysql db-type)
051                  (keyword (str (name id-query) "-mysql"))
052                  id-query)]
053       (if (some? db)
054         (jdbc/with-transaction [tx db]
055           (query-fn tx insert-query params)
056           (get (query-fn tx id-q {}) id-key))
057         (do
058           (query-fn insert-query params)
059           (get (query-fn id-q {}) id-key))))))

063  (defn sqlite->mysql
064    "将 SQLite SQL 转换为 MySQL 兼容 SQL。"
065    [sql]
066    (-> sql
067        ;; SQLite 的 AUTOINCREMENT -> MySQL 的 AUTO_INCREMENT
068        (str/replace #"AUTOINCREMENT" "AUTO_INCREMENT")
069        ;; SQLite 的 INTEGER PRIMARY KEY -> MySQL 的 BIGINT PRIMARY KEY AUTO_INCREMENT
070        (str/replace #"INTEGER PRIMARY KEY" "BIGINT PRIMARY KEY AUTO_INCREMENT")
071        ;; SQLite 的 datetime('now') -> MySQL 的 NOW()
072        (str/replace #"datetime\('now'\)" "NOW()")
073        ;; SQLite 的 date('now') -> MySQL 的 CURDATE()
074        (str/replace #"date\('now'\)" "CURDATE()")
075        ;; SQLite 的 julianday -> MySQL 的 DATEDIFF
076        (str/replace #"julianday\(([^)]+)\)\s*-\s*julianday\(([^)]+)\)" "DATEDIFF($1, $2)")
077        ;; SQLite 的 GROUP_CONCAT -> MySQL 的 GROUP_CONCAT
078        (str/replace #"group_concat" "GROUP_CONCAT")
079        ;; SQLite 的 PRAGMA -> MySQL 的 SHOW
080        (str/replace #"PRAGMA table_info\(([^)]+)\)" "DESCRIBE $1")
081        ;; SQLite 的 sqlite_master -> MySQL 的 information_schema
082        (str/replace #"sqlite_master" "information_schema.tables")
083        ;; SQLite 的 type='table' -> MySQL 的 table_type='BASE TABLE'
084        (str/replace #"type='table'" "table_type='BASE TABLE'")))
085  
086  (defn mysql->sqlite
087    "将 MySQL SQL 转换为 SQLite 兼容 SQL。"
088    [sql]
089    (-> sql
090        ;; MySQL 的 AUTO_INCREMENT -> SQLite 的 AUTOINCREMENT
091        (str/replace #"AUTO_INCREMENT" "AUTOINCREMENT")
092        ;; MySQL 的 BIGINT PRIMARY KEY AUTO_INCREMENT -> SQLite 的 INTEGER PRIMARY KEY
093        (str/replace #"BIGINT PRIMARY KEY AUTO_INCREMENT" "INTEGER PRIMARY KEY")
094        ;; MySQL 的 NOW() -> SQLite 的 datetime('now')
095        (str/replace #"NOW\(\)" "datetime('now')")
096        ;; MySQL 的 CURDATE() -> SQLite 的 date('now')
097        (str/replace #"CURDATE\(\)" "date('now')")
098        ;; MySQL 的 DATEDIFF -> SQLite 的 julianday
099        (str/replace #"DATEDIFF\(([^,]+),\s*([^)]+)\)" "julianday($1) - julianday($2)")
100        ;; MySQL 的 DESCRIBE -> SQLite 的 PRAGMA table_info
101        (str/replace #"DESCRIBE\s+(\w+)" "PRAGMA table_info($1)")))

105  (defn adapt-sql
106    "根据数据库类型适配 SQL。"
107    [db sql]
108    (let [db-type (detect-db-type db)]
109      (case db-type
110        :mysql (sqlite->mysql sql)
111        :sqlite sql
112        sql)))

116  (defn paginate-query
117    "分页查询适配。"
118    [db sql params page-num page-size]
119    (let [db-type (detect-db-type db)
120          offset (* (dec page-num) page-size)
121          paginated-sql (case db-type
122                          :mysql (str sql " LIMIT " page-size " OFFSET " offset)
123                          :sqlite (str sql " LIMIT " page-size " OFFSET " offset)
124                          (str sql " LIMIT " page-size " OFFSET " offset))]
125      (jdbc/execute! (connectable db)
126                     (into [paginated-sql] (vals params))
127                     {:builder-fn rs/as-unqualified-kebab-maps})))

131  (defn get-table-columns
132    "获取表的列信息，返回统一字段：
133     :column_name :data_type :is_nullable :column_default :column_comment
134     :character_maximum_length :numeric_precision :numeric_scale :is_pk"
135    [db table-name]
136    (let [db-type (detect-db-type db)]
137      (case db-type
138        :sqlite
139        (mapv (fn [{:keys [name type notnull dflt_value pk]}]
140                {:column_name name
141                 :data_type type
142                 :is_nullable (if (= 1 notnull) "NO" "YES")
143                 :column_default dflt_value
144                 :dflt_value dflt_value
145                 :column_comment ""
146                 :character_maximum_length nil
147                 :numeric_precision nil
148                 :numeric_scale nil
149                 :is_pk (if (= 1 pk) "YES" "NO")
150                 :pk pk})
151              (jdbc/execute! (connectable db)
152                             [(str "PRAGMA table_info(" table-name ")")]
153                             {:builder-fn rs/as-unqualified-lower-maps}))
154        :mysql
155        (jdbc/execute! (connectable db)
156                       [(str "SELECT c.column_name, c.data_type, c.is_nullable, "
157                             "c.column_default, c.column_comment, "
158                             "c.character_maximum_length, c.numeric_precision, c.numeric_scale, "
159                             "CASE WHEN tc.constraint_type = 'PRIMARY KEY' THEN 'YES' ELSE 'NO' END AS is_pk "
160                             "FROM information_schema.columns c "
161                             "LEFT JOIN information_schema.key_column_usage kcu "
162                             "  ON c.table_schema = kcu.table_schema "
163                             "  AND c.table_name = kcu.table_name "
164                             "  AND c.column_name = kcu.column_name "
165                             "LEFT JOIN information_schema.table_constraints tc "
166                             "  ON kcu.constraint_schema = tc.constraint_schema "
167                             "  AND kcu.constraint_name = tc.constraint_name "
168                             "  AND tc.constraint_type = 'PRIMARY KEY' "
169                             "WHERE c.table_schema = DATABASE() AND c.table_name = ? "
170                             "ORDER BY c.ordinal_position")
171                        table-name]
172                       {:builder-fn rs/as-unqualified-lower-maps})
173        [])))

175  (defn get-tables
176    "获取数据库中的所有表。"
177    [db]
178    (let [db-type (detect-db-type db)]
179      (case db-type
180        :sqlite
181        (jdbc/execute! (connectable db)
182                       ["SELECT name as table_name, COALESCE(name, '') as table_comment FROM sqlite_master WHERE type='table' AND name NOT LIKE 'sqlite_%' ORDER BY name"]
183                       {:builder-fn rs/as-unqualified-lower-maps})
184        :mysql
185        (jdbc/execute! (connectable db)
186                       ["SELECT table_name, IFNULL(table_comment, '') AS table_comment FROM information_schema.tables WHERE table_schema = DATABASE() AND table_type = 'BASE TABLE' ORDER BY table_name"]
187                       {:builder-fn rs/as-unqualified-lower-maps})
188        [])))
