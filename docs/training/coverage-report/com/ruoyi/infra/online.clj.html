001  (ns com.ruoyi.infra.online
002    "在线用户管理。
003  
004    将会话持久化到 sys_online 表，并提供强退/黑名单能力。
005    由于 JWT 令牌在过期前无法单方面失效，强退后会将令牌加入
006    内存黑名单，直到令牌自然过期。"
007    (:require
008     [clojure.tools.logging :as log]
009     [com.ruoyi.infra.security :as security])
010    (:import
011     [java.util.concurrent ScheduledThreadPoolExecutor TimeUnit]))
012  
013  ;; ──────────── 全局状态 ────────────
014  
015  (defonce ^:private query-fn-atom (atom nil))
016  
017  (defonce ^:private token-blacklist
018    ;; token -> 过期时间戳（毫秒）
019    (atom {} :validator map?))
020  
021  (defn set-query-fn!
022    "由 online-service 在系统启动时注入 query-fn。"
023    [query-fn]
024    (reset! query-fn-atom query-fn))
025  
026  (defn- query-fn []
027    (if-let [q @query-fn-atom]
028      q
029      (throw (IllegalStateException. "online query-fn not initialized"))))
030  
031  ;; ──────────── 黑名单 ────────────
032  
033  (defn blacklist!
034    "将令牌加入黑名单，exp-ms 为令牌过期时间戳。"
035    [token exp-ms]
036    (swap! token-blacklist assoc token exp-ms))
037  
038  (defn blacklisted?
039    "检查令牌是否已被强退。"
040    [token]
041    (contains? @token-blacklist token))
042  
043  (defn cleanup-blacklist!
044    "清理已过期的黑名单记录。"
045    []
046    (let [now (System/currentTimeMillis)]
047      (swap! token-blacklist
048             (fn [bl]
049               (into {} (remove (fn [[_ exp]] (< exp now)) bl))))))
050  
051  ;; ──────────── 清理调度 ────────────
052  
053  (declare cleanup-expired-sessions!)
054  
055  (defonce cleanup-executor
056    (delay
057      (doto (ScheduledThreadPoolExecutor. 1)
058        (.scheduleAtFixedRate
059         (reify Runnable
060           (run [_]
061             (try
062               (cleanup-expired-sessions!)
063               (cleanup-blacklist!)
064               (catch Exception e
065                 (log/warn e "Online user cleanup failed")))))
066         5 5 TimeUnit/MINUTES))))
067  
068  ;; ──────────── 核心 API ────────────
069  
070  (defn register!
071    "登录成功后注册在线用户记录。"
072    [token user-name login-ip]
073    (let [now (System/currentTimeMillis)
074          expire-ms (* 30 60 1000)]  ; 30 分钟无访问视为过期
075      (try
076        ((query-fn) :create-online-user!
077                    {:session_id token
078                     :login_name user-name
079                     :dept_name ""
080                     :ipaddr (or login-ip "127.0.0.1")
081                     :login_location ""
082                     :browser ""
083                     :os ""
084                     :status "on_line"
085                     :start_timestamp now
086                     :last_access_time now
087                     :expire_time expire-ms})
088        (catch Exception e
089          (log/warn e "Failed to register online user")))
090      (force cleanup-executor)
091      nil))
092  
093  (defn heartbeat!
094    "更新用户最后访问时间。"
095    [token]
096    (when token
097      (try
098        ((query-fn) :update-online-user!
099                    {:session_id token
100                     :last_access_time (System/currentTimeMillis)
101                     :status nil
102                     :expire_time nil})
103        (catch Exception e
104          (log/warn e "Failed to update online user heartbeat"))))
105    nil)
106  
107  (defn unregister!
108    "注销在线用户（用户主动退出）。"
109    [token]
110    (when token
111      (try
112        ((query-fn) :delete-online-user! {:session_id token})
113        (catch Exception e
114          (log/warn e "Failed to unregister online user")))
115      (when-let [claims (some-> token security/parse-token)]
116        (blacklist! token (:exp claims))))
117    nil)
118  
119  (defn cleanup-expired-sessions!
120    "清理超过 expire_time 未心跳的会话。"
121    []
122    (let [threshold (- (System/currentTimeMillis) (* 30 60 1000))]
123      (try
124        ((query-fn) :delete-online-user! {:session_id "__cleanup__"})
125        (catch Exception _))
126      ;; 由于 HugSQL 没有动态 WHERE，这里简单列出后逐条删除
127      (let [expired (->> ((query-fn) :list-online-users {:page_size 10000 :offset 0})
128                         (filter #(< (:last_access_time %) threshold)))]
129        (doseq [s expired]
130          (try
131            ((query-fn) :delete-online-user! {:session_id (:session_id s)})
132            (catch Exception e
133              (log/warn e "Failed to delete expired session"))))))
134    nil)
135  
136  (defn list-online
137    "获取在线用户列表，支持条件筛选。"
138    [& {:keys [login-name ipaddr page-num page-size]
139        :or   {page-num 1 page-size 10}}]
140    (let [offset (* (dec page-num) page-size)
141          rows ((query-fn) :list-online-users
142                           {:ipaddr ipaddr
143                            :login_name login-name
144                            :page_size page-size
145                            :offset offset})
146          total (:total ((query-fn) :count-online-users
147                                    {:ipaddr ipaddr
148                                     :login_name login-name}))]
149      {:rows (mapv #(-> %
150                        (assoc :token-id (:session_id %))
151                        (assoc :token (:session_id %))
152                        (assoc :user-id (:login_name %))
153                        (assoc :user-name (:login_name %))
154                        (assoc :login-ip (:ipaddr %))
155                        (assoc :login-time (:start_timestamp %))
156                        (assoc :last-access (:last_access_time %)))
157                   rows)
158       :total total}))
159  
160  (defn force-logout!
161    "强退指定在线用户。"
162    [token]
163    (when token
164      (try
165        ((query-fn) :delete-online-user! {:session_id token})
166        (catch Exception e
167          (log/warn e "Failed to force logout online user")))
168      (when-let [claims (some-> token security/parse-token)]
169        (blacklist! token (:exp claims))))
170    {:success true})
